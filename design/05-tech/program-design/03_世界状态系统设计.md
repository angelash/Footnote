## 1. 系统概述

### 1.1 职责定义

WorldState是游戏的核心状态管理系统，负责：

1. **计数器管理** - R/P/W三大隐藏计数器的维护和计算
2. **标记管理** - 全局游戏标记（flags）的读写
3. **Zone状态管理** - 各Zone的解锁、访问、完成状态
4. **伤痕管理** - 深度介入产生的永久伤痕记录
5. **污染管理** - 时间回溯产生的污染效果
6. **序列化** - 状态的存储和恢复

### 1.2 设计原则

| 原则 | 说明 |
|------|------|
| **单一数据源** | 所有游戏状态由WorldState统一管理 |
| **不可变更新** | 状态变更通过方法调用，不直接修改 |
| **事件驱动** | 状态变更触发事件通知订阅者 |
| **可序列化** | 所有状态可完整序列化用于存档 |
| **可追溯** | 关键状态变更记录历史 |

## 2. 计数器系统

### 2.1 计数器定义

| 计数器 | 范围 | 说明 | 可见性 |
|--------|------|------|--------|
| **R** | 0-15 | 无收益残差 - 玩家"无意义"行为累积 | 完全隐藏 |
| **P** | 0-20 | 观察者压力 - 高维能力使用累积 | 完全隐藏 |
| **W** | 0-100 | 世界可读性 - 由R、P、事件计算得出 | 部分显影 |

### 2.2 CounterManager实现

```typescript
// src/systems/world/CounterManager.ts

import { ICounters, TCounterType, ICounterThreshold } from '@/types';
import { eventBus, GameEvent } from '@/systems/EventBus';

/**
 * 计数器配置
 */
const COUNTER_CONFIG = {
  R: { min: 0, max: 15, default: 0 },
  P: { min: 0, max: 20, default: 0 },
  W: { min: 0, max: 100, default: 100 },
};

/**
 * R值阈值定义
 */
const R_THRESHOLDS: ICounterThreshold[] = [
  { type: 'R', level: 'subtle', value: 3, effects: ['dialogue_pause'] },
  { type: 'R', level: 'judgment', value: 6, effects: ['judgment_sentence'] },
  { type: 'R', level: 'rewrite', value: 10, effects: ['model_rewrite', 'ending_c_unlock'] },
];

/**
 * P值阈值定义
 */
const P_THRESHOLDS: ICounterThreshold[] = [
  { type: 'P', level: 'light', value: 4, effects: ['npc_mechanical'] },
  { type: 'P', level: 'moderate', value: 7, effects: ['env_patches', 'options_reduce'] },
  { type: 'P', level: 'strong', value: 11, effects: ['system_accounting'] },
  { type: 'P', level: 'extreme', value: 16, effects: ['system_repair'] },
];

/**
 * W值阈值定义
 */
const W_THRESHOLDS: ICounterThreshold[] = [
  { type: 'W', level: 'unstable', value: 60, effects: ['screen_shake'] },
  { type: 'W', level: 'critical', value: 40, effects: ['color_shift', 'ui_glitch'] },
  { type: 'W', level: 'collapse', value: 20, effects: ['layer3_reveal'] },
];

/**
 * 计数器管理器
 */
export class CounterManager {
  private _R: number = COUNTER_CONFIG.R.default;
  private _P: number = COUNTER_CONFIG.P.default;
  private _eventModifier: number = 0;
  private _triggeredThresholds: Set<string> = new Set();
  private _changeHistory: Array<{
    type: TCounterType;
    oldValue: number;
    newValue: number;
    source: string;
    timestamp: number;
  }> = [];

  /**
   * 获取R值
   */
  get R(): number {
    return this._R;
  }

  /**
   * 获取P值
   */
  get P(): number {
    return this._P;
  }

  /**
   * 获取W值（计算值）
   */
  get W(): number {
    return this._calculateW();
  }

  /**
   * 获取所有计数器
   */
  getAll(): ICounters {
    return {
      R: this._R,
      P: this._P,
      W: this.W,
    };
  }

  /**
   * 增加R值
   * @param amount 增加量（可为负）
   * @param source 来源标识
   */
  addR(amount: number, source: string): void {
    const oldValue = this._R;
    const newValue = this._clamp(
      this._R + amount,
      COUNTER_CONFIG.R.min,
      COUNTER_CONFIG.R.max
    );

    if (oldValue !== newValue) {
      this._R = newValue;
      this._recordChange('R', oldValue, newValue, source);
      this._emitChange('R', oldValue, newValue);
      this._checkThresholds('R', newValue);
      // R变化会影响W
      this._onWChange();
    }
  }

  /**
   * 增加P值
   * @param amount 增加量（可为负）
   * @param source 来源标识
   */
  addP(amount: number, source: string): void {
    const oldValue = this._P;
    const newValue = this._clamp(
      this._P + amount,
      COUNTER_CONFIG.P.min,
      COUNTER_CONFIG.P.max
    );

    if (oldValue !== newValue) {
      this._P = newValue;
      this._recordChange('P', oldValue, newValue, source);
      this._emitChange('P', oldValue, newValue);
      this._checkThresholds('P', newValue);
      // P变化会影响W
      this._onWChange();
    }
  }

  /**
   * 设置事件修正值
   * @param value 修正值（累加到现有值）
   */
  addEventModifier(value: number): void {
    this._eventModifier += value;
    this._onWChange();
  }

  /**
   * 重置事件修正值
   */
  resetEventModifier(): void {
    this._eventModifier = 0;
    this._onWChange();
  }

  /**
   * 计算W值
   * W = 100 - (R × 3) - (P × 2) - eventModifier
   */
  private _calculateW(): number {
    const rawW = 100 - (this._R * 3) - (this._P * 2) - this._eventModifier;
    return this._clamp(rawW, COUNTER_CONFIG.W.min, COUNTER_CONFIG.W.max);
  }

  /**
   * W值变化处理
   */
  private _onWChange(): void {
    const newW = this.W;
    this._checkThresholds('W', newW);
    eventBus.emit(GameEvent.W_CHANGE, {
      type: 'W',
      value: newW,
    });
  }

  /**
   * 检查阈值触发
   */
  private _checkThresholds(type: TCounterType, value: number): void {
    const thresholds = this._getThresholds(type);
    
    for (const threshold of thresholds) {
      const thresholdKey = `${type}_${threshold.level}`;
      const shouldTrigger = type === 'W'
        ? value <= threshold.value
        : value >= threshold.value;

      if (shouldTrigger && !this._triggeredThresholds.has(thresholdKey)) {
        this._triggeredThresholds.add(thresholdKey);
        eventBus.emit(GameEvent.THRESHOLD_REACHED, {
          type,
          level: threshold.level,
          value,
          effects: threshold.effects,
        });
      }
    }
  }

  /**
   * 获取对应类型的阈值配置
   */
  private _getThresholds(type: TCounterType): ICounterThreshold[] {
    switch (type) {
      case 'R': return R_THRESHOLDS;
      case 'P': return P_THRESHOLDS;
      case 'W': return W_THRESHOLDS;
    }
  }

  /**
   * 记录变更历史
   */
  private _recordChange(
    type: TCounterType,
    oldValue: number,
    newValue: number,
    source: string
  ): void {
    this._changeHistory.push({
      type,
      oldValue,
      newValue,
      source,
      timestamp: Date.now(),
    });

    // 限制历史记录数量
    if (this._changeHistory.length > 1000) {
      this._changeHistory.shift();
    }
  }

  /**
   * 发送变更事件
   */
  private _emitChange(type: TCounterType, oldValue: number, newValue: number): void {
    eventBus.emit(GameEvent.COUNTER_CHANGE, {
      type,
      oldValue,
      newValue,
    });

    // 发送特定类型事件
    const specificEvent = type === 'R' ? GameEvent.R_CHANGE : GameEvent.P_CHANGE;
    eventBus.emit(specificEvent, {
      type,
      oldValue,
      newValue,
    });
  }

  /**
   * 数值限制
   */
  private _clamp(value: number, min: number, max: number): number {
    return Math.max(min, Math.min(max, value));
  }

  /**
   * 序列化
   */
  serialize(): { R: number; P: number; eventModifier: number; triggeredThresholds: string[] } {
    return {
      R: this._R,
      P: this._P,
      eventModifier: this._eventModifier,
      triggeredThresholds: Array.from(this._triggeredThresholds),
    };
  }

  /**
   * 反序列化
   */
  deserialize(data: { R: number; P: number; eventModifier: number; triggeredThresholds: string[] }): void {
    this._R = data.R;
    this._P = data.P;
    this._eventModifier = data.eventModifier;
    this._triggeredThresholds = new Set(data.triggeredThresholds);
  }

  /**
   * 重置
   */
  reset(): void {
    this._R = COUNTER_CONFIG.R.default;
    this._P = COUNTER_CONFIG.P.default;
    this._eventModifier = 0;
    this._triggeredThresholds.clear();
    this._changeHistory = [];
  }

  /**
   * 获取变更历史（调试用）
   */
  getChangeHistory(): typeof this._changeHistory {
    return [...this._changeHistory];
  }
}
```

```typescript
// src/systems/world/FlagManager.ts

import { eventBus, GameEvent } from '@/systems/EventBus';

/**
 * 标记分类
 */
export enum FlagCategory {
  /** 剧情进度 */
  STORY = 'story',
  /** 对话选择 */
  DIALOGUE = 'dialogue',
  /** 物品获取 */
  ITEM = 'item',
  /** 能力解锁 */
  ABILITY = 'ability',
  /** 伏笔触发 */
  FORESHADOW = 'foreshadow',
  /** 系统标记 */
  SYSTEM = 'system',
}

/**
 * 标记命名约定
 * 格式: {category}_{chapter}_{description}
 * 示例: story_c1_met_gulin, dialogue_c2_questioned_system
 */
export class FlagManager {
  private _flags: Map<string, boolean> = new Map();
  private _changeListeners: Map<string, Array<(value: boolean) => void>> = new Map();

  /**
   * 获取标记值
   * @param key 标记名
   * @returns 标记值，未设置返回false
   */
  get(key: string): boolean {
    return this._flags.get(key) ?? false;
  }

  /**
   * 设置标记
   * @param key 标记名
   * @param value 标记值
   */
  set(key: string, value: boolean): void {
    const oldValue = this.get(key);
    
    if (oldValue !== value) {
      this._flags.set(key, value);
      
      // 触发事件
      eventBus.emit(GameEvent.FLAG_CHANGE, {
        key,
        oldValue,
        newValue: value,
      });

      // 触发特定标记监听
      const listeners = this._changeListeners.get(key);
      if (listeners) {
        listeners.forEach(fn => fn(value));
      }
    }
  }

  /**
   * 切换标记
   * @param key 标记名
   */
  toggle(key: string): void {
    this.set(key, !this.get(key));
  }

  /**
   * 检查标记是否存在
   * @param key 标记名
   */
  has(key: string): boolean {
    return this._flags.has(key);
  }

  /**
   * 批量设置标记
   * @param flags 标记对象
   */
  setMultiple(flags: Record<string, boolean>): void {
    for (const [key, value] of Object.entries(flags)) {
      this.set(key, value);
    }
  }

  /**
   * 获取所有标记
   */
  getAll(): Record<string, boolean> {
    return Object.fromEntries(this._flags);
  }

  /**
   * 按分类获取标记
   * @param category 分类
   */
  getByCategory(category: FlagCategory): Record<string, boolean> {
    const result: Record<string, boolean> = {};
    
    for (const [key, value] of this._flags) {
      if (key.startsWith(category)) {
        result[key] = value;
      }
    }
    
    return result;
  }

  /**
   * 监听特定标记变化
   * @param key 标记名
   * @param callback 回调函数
   * @returns 取消监听函数
   */
  watch(key: string, callback: (value: boolean) => void): () => void {
    if (!this._changeListeners.has(key)) {
      this._changeListeners.set(key, []);
    }
    
    this._changeListeners.get(key)!.push(callback);
    
    return () => {
      const listeners = this._changeListeners.get(key);
      if (listeners) {
        const index = listeners.indexOf(callback);
        if (index > -1) {
          listeners.splice(index, 1);
        }
      }
    };
  }

  /**
   * 检查多个标记（AND逻辑）
   * @param keys 标记名数组
   */
  checkAll(keys: string[]): boolean {
    return keys.every(key => this.get(key));
  }

  /**
   * 检查多个标记（OR逻辑）
   * @param keys 标记名数组
   */
  checkAny(keys: string[]): boolean {
    return keys.some(key => this.get(key));
  }

  /**
   * 序列化
   */
  serialize(): Record<string, boolean> {
    return this.getAll();
  }

  /**
   * 反序列化
   */
  deserialize(data: Record<string, boolean>): void {
    this._flags.clear();
    for (const [key, value] of Object.entries(data)) {
      this._flags.set(key, value);
    }
  }

  /**
   * 重置
   */
  reset(): void {
    this._flags.clear();
    this._changeListeners.clear();
  }

  /**
   * 获取标记数量
   */
  get count(): number {
    return this._flags.size;
  }
}
```

```typescript
// src/systems/world/ZoneStateManager.ts

import { TZoneId, IZoneStateData, ZoneState } from '@/types';
import { eventBus, GameEvent } from '@/systems/EventBus';

/**
 * Zone配置（解锁条件等）
 */
interface IZoneConfig {
  id: TZoneId;
  chapter: string;
  unlockConditions: string[];  // 需要满足的标记
  previousZone?: TZoneId;      // 前置Zone
}

/**
 * Zone状态管理器
 */
export class ZoneStateManager {
  private _states: Map<TZoneId, IZoneStateData> = new Map();
  private _currentZone: TZoneId | null = null;
  private _visitOrder: TZoneId[] = [];

  /**
   * 获取Zone状态
   */
  get(zoneId: TZoneId): IZoneStateData {
    if (!this._states.has(zoneId)) {
      this._states.set(zoneId, this._createDefaultState());
    }
    return this._states.get(zoneId)!;
  }

  /**
   * 设置Zone状态
   */
  setState(zoneId: TZoneId, state: ZoneState): void {
    const zoneState = this.get(zoneId);
    const oldState = zoneState.state;
    
    if (oldState !== state) {
      zoneState.state = state;
      
      eventBus.emit(GameEvent.ZONE_STATE_CHANGE, {
        zoneId,
        oldState,
        newState: state,
      });
    }
  }

  /**
   * 解锁Zone
   */
  unlock(zoneId: TZoneId): void {
    const zoneState = this.get(zoneId);
    
    if (zoneState.state === ZoneState.LOCKED) {
      this.setState(zoneId, ZoneState.UNLOCKED);
      
      eventBus.emit(GameEvent.ZONE_UNLOCKED, { zoneId });
    }
  }

  /**
   * 访问Zone
   */
  visit(zoneId: TZoneId): void {
    const zoneState = this.get(zoneId);
    const now = Date.now();
    const isFirstVisit = !zoneState.firstVisitAt;
    
    // 更新访问数据
    if (isFirstVisit) {
      zoneState.firstVisitAt = now;
    }
    zoneState.lastVisitAt = now;
    zoneState.visitCount++;
    
    // 状态流转
    if (zoneState.state === ZoneState.UNLOCKED) {
      this.setState(zoneId, ZoneState.VISITED);
    }
    
    // 记录访问顺序
    this._visitOrder.push(zoneId);
    this._currentZone = zoneId;
    
    // 发送事件
    eventBus.emit(GameEvent.ZONE_ENTER, {
      zoneId,
      isFirstVisit,
      isRevisit: !isFirstVisit,
    });
  }

  /**
   * 完成Zone
   */
  complete(zoneId: TZoneId): void {
    const zoneState = this.get(zoneId);
    
    if (zoneState.state !== ZoneState.COMPLETED) {
      this.setState(zoneId, ZoneState.COMPLETED);
      
      eventBus.emit(GameEvent.ZONE_COMPLETE, { zoneId });
    }
  }

  /**
   * 离开Zone
   */
  exit(fromZoneId: TZoneId, toZoneId: TZoneId): void {
    eventBus.emit(GameEvent.ZONE_EXIT, {
      zoneId: fromZoneId,
      nextZoneId: toZoneId,
    });
  }

  /**
   * 检查Zone是否解锁
   */
  isUnlocked(zoneId: TZoneId): boolean {
    const state = this.get(zoneId).state;
    return state !== ZoneState.LOCKED;
  }

  /**
   * 检查Zone是否已访问
   */
  isVisited(zoneId: TZoneId): boolean {
    const state = this.get(zoneId).state;
    return state === ZoneState.VISITED || state === ZoneState.COMPLETED;
  }

  /**
   * 检查Zone是否已完成
   */
  isCompleted(zoneId: TZoneId): boolean {
    return this.get(zoneId).state === ZoneState.COMPLETED;
  }

  /**
   * 获取当前Zone
   */
  getCurrentZone(): TZoneId | null {
    return this._currentZone;
  }

  /**
   * 获取访问顺序
   */
  getVisitOrder(): TZoneId[] {
    return [...this._visitOrder];
  }

  /**
   * 记录Zone内获得的卡片
   */
  addObtainedCard(zoneId: TZoneId, cardId: string): void {
    const zoneState = this.get(zoneId);
    if (!zoneState.obtainedCards.includes(cardId)) {
      zoneState.obtainedCards.push(cardId);
    }
  }

  /**
   * 记录Zone内触发的伏笔
   */
  addTriggeredForeshadow(zoneId: TZoneId, foreshadowId: string): void {
    const zoneState = this.get(zoneId);
    if (!zoneState.triggeredForeshadows.includes(foreshadowId)) {
      zoneState.triggeredForeshadows.push(foreshadowId);
    }
  }

  /**
   * 标记使用过深度能力
   */
  markDepthAbilityUsed(zoneId: TZoneId): void {
    this.get(zoneId).usedDepthAbility = true;
  }

  /**
   * 创建默认Zone状态
   */
  private _createDefaultState(): IZoneStateData {
    return {
      state: ZoneState.LOCKED,
      visitCount: 0,
      usedDepthAbility: false,
      obtainedCards: [],
      triggeredForeshadows: [],
    };
  }

  /**
   * 序列化
   */
  serialize(): Record<TZoneId, IZoneStateData> {
    const result: Record<string, IZoneStateData> = {};
    for (const [id, state] of this._states) {
      result[id] = { ...state };
    }
    return result as Record<TZoneId, IZoneStateData>;
  }

  /**
   * 反序列化
   */
  deserialize(data: Record<TZoneId, IZoneStateData>): void {
    this._states.clear();
    for (const [id, state] of Object.entries(data)) {
      this._states.set(id as TZoneId, { ...state });
    }
  }

  /**
   * 重置
   */
  reset(): void {
    this._states.clear();
    this._currentZone = null;
    this._visitOrder = [];
  }

  /**
   * 获取章节完成度
   */
  getChapterProgress(chapter: string): { completed: number; total: number } {
    let completed = 0;
    let total = 0;
    
    for (const [id, state] of this._states) {
      if (id.startsWith(chapter)) {
        total++;
        if (state.state === ZoneState.COMPLETED) {
          completed++;
        }
      }
    }
    
    return { completed, total };
  }
}
```

```typescript
   * 获取受影响的Zone列表
   */
  getAffectedZones(): TZoneId[] {
    const zones = new Set<TZoneId>();
    
    for (const c of this._contaminations) {
      c.affectedZones.forEach(z => zones.add(z));
    }
    
    return Array.from(zones);
  }

  /**
   * 检查Zone是否受污染
   */
  isZoneContaminated(zoneId: TZoneId): boolean {
    return this._contaminations.some(c => c.affectedZones.includes(zoneId));
  }

  /**
   * 获取Zone的污染强度
   */
  getZoneContaminationLevel(zoneId: TZoneId): number {
    let totalIntensity = 0;
    
    for (const c of this._contaminations) {
      if (c.affectedZones.includes(zoneId)) {
        totalIntensity += c.intensity;
      }
    }
    
    return Math.min(1, totalIntensity);  // 限制最大值为1
  }

  /**
   * 获取总污染等级
   */
  getTotalContaminationLevel(): number {
    if (this._contaminations.length === 0) return 0;
    
    const total = this._contaminations.reduce((sum, c) => sum + c.intensity, 0);
    return Math.min(1, total / 3);  // 归一化
  }

  /**
   * 序列化
   */
  serialize(): IContamination[] {
    return [...this._contaminations];
  }

  /**
   * 反序列化
   */
  deserialize(data: IContamination[]): void {
    this._contaminations = [...data];
    const maxId = Math.max(0, ...data.map(c => parseInt(c.id.split('_')[1]) || 0));
    this._idCounter = maxId;
  }

  /**
   * 重置
   */
  reset(): void {
    this._contaminations = [];
    this._idCounter = 0;
  }
```

```typescript
// src/systems/world/WorldState.ts

import { 
  IWorldStateData, 
  ICounters, 
  TZoneId, 
  IZoneStateData, 
  ZoneState,
  IScar,
  IContamination,
  IConditionConfig,
  ISerializable
} from '@/types';
import { CounterManager } from './CounterManager';
import { FlagManager } from './FlagManager';
import { ZoneStateManager } from './ZoneStateManager';
import { ScarManager } from './ScarManager';
import { ContaminationManager } from './ContaminationManager';
import { eventBus, GameEvent } from '@/systems/EventBus';

/**
 * 世界状态管理器 - 单例
 */
export class WorldState implements ISerializable<IWorldStateData> {
  private static _instance: WorldState;
  
  private _counters: CounterManager;
  private _flags: FlagManager;
  private _zones: ZoneStateManager;
  private _scars: ScarManager;
  private _contaminations: ContaminationManager;
  private _currentChapter: string = 'C0';
  private _initialized: boolean = false;

  private constructor() {
    this._counters = new CounterManager();
    this._flags = new FlagManager();
    this._zones = new ZoneStateManager();
    this._scars = new ScarManager();
    this._contaminations = new ContaminationManager();
  }

  /**
   * 获取单例实例
   */
  static getInstance(): WorldState {
    if (!WorldState._instance) {
      WorldState._instance = new WorldState();
    }
    return WorldState._instance;
  }

  /**
   * 初始化
   */
  initialize(): void {
    if (this._initialized) return;
    
    // 初始解锁序章第一个Zone
    this._zones.unlock('C0-Z1' as TZoneId);
    
    this._initialized = true;
    eventBus.emit(GameEvent.WORLD_STATE_INIT);

  /**
   * 重置
   */

    this._counters.reset();
    this._flags.reset();
    this._zones.reset();
    this._scars.reset();
    this._contaminations.reset();
    this._currentChapter = 'C0';
    this._initialized = false;
  }

  // ==================== 计数器接口 ====================

  /**
   * 获取R值
   */
  getR(): number {
    return this._counters.R;
  }

  /**
   * 获取P值
   */
  getP(): number {
    return this._counters.P;
  }

  /**
   * 获取W值
   */
  getW(): number {
    return this._counters.W;
  }

  /**
   * 获取所有计数器
   */
  getCounters(): ICounters {
    return this._counters.getAll();
  }

  /**
   * 增加R值
   */
  addR(amount: number, source: string = 'unknown'): void {
    this._counters.addR(amount, source);
  }

  /**
   * 增加P值
   */
  addP(amount: number, source: string = 'unknown'): void {
    this._counters.addP(amount, source);
  }

  /**
   * 增加事件修正值
   */

    this._counters.addEventModifier(value);
  }

  // ==================== 标记接口 ====================

   * 获取标记

    return this._flags.get(key);

  }

   */

    return this._scars.getByZone(zoneId);
  }

  /**
   * 获取所有伤痕
   */
  getAllScars(): IScar[] {
    return this._scars.getAll();
  }

  // ==================== 污染接口 ====================

  /**
   * 添加污染
   */
  addContamination(params: Omit<IContamination, 'id' | 'createdAt'>): IContamination {
    return this._contaminations.add(params);
  }

  /**
   * 获取污染等级
   */
  getContaminationLevel(): number {
    return this._contaminations.getTotalContaminationLevel();
  }

  /**
   * 获取受污染的Zone
   */
  getContaminatedZones(): TZoneId[] {
    return this._contaminations.getAffectedZones();
  }

  // ==================== 条件检查 ====================

  /**
   * 检查条件是否满足
   */
  checkCondition(condition: IConditionConfig): boolean {
    switch (condition.type) {
      case 'flag':
        return this.getFlag(condition.flag!) === (condition.flagValue ?? true);
      
      case 'counter':
        const counterValue = this._getCounterValue(condition.counter!);
        return this._compareValue(counterValue, condition.operator!, condition.value!);
      
      case 'ability':
        return this.getFlag(`ability_${condition.ability}_unlocked`);
      
      case 'card':
        return this.getFlag(`card_${condition.card}_obtained`);
      
      case 'zone':
        if (condition.zone!.state === 'visited') {
          return this.isZoneVisited(condition.zone!.id);
        }
        return this.isZoneCompleted(condition.zone!.id);
      
      case 'and':
        return condition.conditions!.every(c => this.checkCondition(c));
      
      case 'or':
        return condition.conditions!.some(c => this.checkCondition(c));
      
      case 'not':
        return !this.checkCondition(condition.conditions![0]);
      
      default:
        return false;
    }
  }

  private _getCounterValue(type: 'R' | 'P' | 'W'): number {
    switch (type) {
      case 'R': return this.getR();
      case 'P': return this.getP();
      case 'W': return this.getW();
    }
  }

  private _compareValue(value: number, operator: string, target: number): boolean {
    switch (operator) {
      case '<': return value < target;
      case '<=': return value <= target;
      case '=': return value === target;
      case '>=': return value >= target;
      case '>': return value > target;
      default: return false;
    }
  }

  // ==================== 序列化 ====================

  /**
   * 序列化
   */
  serialize(): IWorldStateData {
    return {
      counters: this._counters.getAll(),
      abilities: this._getUnlockedAbilities(),
      flags: this._flags.getAll(),
      zoneStates: this._zones.serialize(),
      scars: this._scars.serialize(),
      contaminations: this._contaminations.serialize(),
      eventModifier: 0, // TODO: 从CounterManager获取
      currentChapter: this._currentChapter,
      currentZone: this._zones.getCurrentZone() || ('C0-Z1' as TZoneId),
    };
  }

  /**
   * 反序列化
   */
  deserialize(data: IWorldStateData): void {
    this._counters.deserialize({
      R: data.counters.R,
      P: data.counters.P,
      eventModifier: data.eventModifier,
      triggeredThresholds: [],
    });
    
    this._flags.deserialize(data.flags);
    this._zones.deserialize(data.zoneStates);
    this._scars.deserialize(data.scars);
    this._contaminations.deserialize(data.contaminations);
    this._currentChapter = data.currentChapter;
    
    this._initialized = true;
  }

  private _getUnlockedAbilities(): string[] {
    const abilities: string[] = [];
    if (this.getFlag('ability_perception_unlocked')) abilities.push('depth_perception');
    if (this.getFlag('ability_intervention_unlocked')) abilities.push('depth_intervention');
    if (this.getFlag('ability_time_unlocked')) abilities.push('time_intervention');
    return abilities;
  }
}

export const worldState = WorldState.getInstance();
```
