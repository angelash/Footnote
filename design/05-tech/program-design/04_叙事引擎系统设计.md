## 1. 系统概述

### 1.1 职责定义

NarrativeEngine是游戏的叙事核心，负责：

1. **对话管理** - 对话加载、播放、选择处理
2. **卡片系统** - 卡片获取、存储、展示
3. **伏笔追踪** - 伏笔的投放、加深、回收状态管理
4. **角色管理** - 角色数据、立绘、表情管理

### 2.1 DialogueManager实现

```typescript
// src/systems/narrative/DialogueManager.ts

import { 
  IDialogue, 
  IDialogueLine, 
  IDialogueOption, 
  IDialogueState,
  IDialogueEffects,
  IDialogueManagerConfig 
} from '@/types';
import { eventBus, GameEvent } from '@/systems/EventBus';
import { worldState } from '@/systems/world';

/**
 * 默认配置
 */
const DEFAULT_CONFIG: IDialogueManagerConfig = {
  defaultSpeed: 30,      // 30字/秒
  allowSkip: true,
  autoDelay: 1500,       // 1.5秒
  historyLimit: 100,
};

/**
 * 对话管理器
 */
export class DialogueManager {
  private _dialogues: Map<string, IDialogue> = new Map();
  private _state: IDialogueState;
  private _config: IDialogueManagerConfig;
  private _currentDialogue: IDialogue | null = null;
  private _onLineCallback?: (line: IDialogueLine) => void;
  private _onOptionsCallback?: (options: IDialogueOption[]) => void;
  private _onEndCallback?: () => void;

  constructor(config: Partial<IDialogueManagerConfig> = {}) {
    this._config = { ...DEFAULT_CONFIG, ...config };
    this._state = this._createInitialState();
  }

  /**
   * 注册对话数据
   */
  registerDialogues(dialogues: Record<string, IDialogue>): void {
    for (const [id, dialogue] of Object.entries(dialogues)) {
      this._dialogues.set(id, dialogue);
    }
  }

  /**
   * 加载对话
   */
  async load(dialogueId: string): Promise<IDialogue | null> {
    // 先检查缓存
    if (this._dialogues.has(dialogueId)) {
      return this._dialogues.get(dialogueId)!;
    }
    
    // TODO: 从YAML文件动态加载
    console.warn(`Dialogue ${dialogueId} not found`);
    return null;
  }

  /**
   * 开始对话
   */
  async start(dialogueId: string): Promise<boolean> {
    const dialogue = await this.load(dialogueId);
    if (!dialogue) return false;

    this._currentDialogue = dialogue;
    this._state = {
      currentDialogueId: dialogueId,
      currentLineIndex: 0,
      isTyping: false,
      isWaitingInput: false,
      optionsVisible: false,
      history: [],
    };

    eventBus.emit(GameEvent.DIALOGUE_START, {
      dialogueId,
      speaker: dialogue.speaker,
    });

    // 播放第一行
    await this._playCurrentLine();
    
    return true;
  }

  /**
   * 推进对话
   */
  async advance(): Promise<void> {
    if (!this._currentDialogue) return;

    // 如果正在打字，完成当前行
    if (this._state.isTyping) {
      this._completeTyping();
      return;
    }

    // 如果显示选项，等待选择
    if (this._state.optionsVisible) {
      return;
    }

    // 检查是否有下一行
    const nextIndex = this._state.currentLineIndex + 1;
    
    if (nextIndex < this._currentDialogue.lines.length) {
      // 播放下一行
      this._state.currentLineIndex = nextIndex;
      await this._playCurrentLine();
    } else {
      // 检查是否有选项
      if (this._currentDialogue.options && this._currentDialogue.options.length > 0) {
        this._showOptions();
      } else {
        // 结束对话
        this._endDialogue();
      }
    }
  }

  /**
   * 选择选项
   */
  async selectChoice(choiceId: string): Promise<void> {
    if (!this._currentDialogue || !this._state.optionsVisible) return;

    const option = this._currentDialogue.options?.find(o => o.id === choiceId);
    if (!option) return;

    // 记录选择
    this._recordHistory(
      this._currentDialogue.speaker,
      `[选择] ${option.text}`,
      choiceId
    );

    // 隐藏选项
    this._state.optionsVisible = false;

    // 发送事件
    eventBus.emit(GameEvent.DIALOGUE_CHOICE, {
      choiceId,
      dialogueId: this._state.currentDialogueId!,
    });

    // 处理选择效果
    if (option.effects) {
      this._applyEffects(option.effects);
    }

    // 跳转到下一个对话
    if (option.next) {
      await this.start(option.next);
    } else {
      this._endDialogue();
    }
  }

  /**
   * 获取当前状态
   */
  getState(): IDialogueState {
    return { ...this._state };
  }

  /**
   * 设置回调
   */
  setCallbacks(callbacks: {
    onLine?: (line: IDialogueLine) => void;
    onOptions?: (options: IDialogueOption[]) => void;
    onEnd?: () => void;
  }): void {
    this._onLineCallback = callbacks.onLine;
    this._onOptionsCallback = callbacks.onOptions;
    this._onEndCallback = callbacks.onEnd;
  }

  /**
   * 跳过当前对话
   */
  skip(): void {
    if (this._config.allowSkip && this._currentDialogue) {
      this._endDialogue();
    }
  }

  /**
   * 是否正在进行对话
   */
  isActive(): boolean {
    return this._currentDialogue !== null;
  }

  // ==================== 私有方法 ====================

  private async _playCurrentLine(): Promise<void> {
    if (!this._currentDialogue) return;

    const line = this._currentDialogue.lines[this._state.currentLineIndex];
    const speaker = line.speaker || this._currentDialogue.speaker;

    // 记录历史
    this._recordHistory(speaker, line.text);

    // 发送行事件
    eventBus.emit(GameEvent.DIALOGUE_LINE, {
      dialogueId: this._state.currentDialogueId,
      lineIndex: this._state.currentLineIndex,
      speaker,
      text: line.text,
    });

    // 回调UI
    this._onLineCallback?.(line);

    // 播放音效
    if (line.sfx) {
      eventBus.emit(GameEvent.SFX_PLAY, { key: line.sfx });
    }

    // 处理回调
    if (line.callback) {
      this._handleCallback(line.callback);
    }

    // 开始打字效果
    this._state.isTyping = true;

    // 模拟打字完成（实际打字在UI层处理）
    const speed = line.speed || this._config.defaultSpeed;
    const duration = (line.text.length / speed) * 1000;

    await this._wait(duration);

    this._state.isTyping = false;

    // 自动播放处理
    if (line.auto) {
      const delay = line.delay || this._config.autoDelay;
      await this._wait(delay);
      await this.advance();
    } else {
      this._state.isWaitingInput = true;
    }
  }

  private _completeTyping(): void {
    this._state.isTyping = false;
    // 触发UI完成打字
    eventBus.emit(GameEvent.DIALOGUE_TYPING_COMPLETE);
  }

  private _showOptions(): void {
    if (!this._currentDialogue?.options) return;

    // 过滤满足条件的选项
    const availableOptions = this._currentDialogue.options.filter(option => {
      if (!option.condition) return true;
      return worldState.checkCondition(option.condition);
    });

    this._state.optionsVisible = true;
    this._onOptionsCallback?.(availableOptions);

    eventBus.emit(GameEvent.DIALOGUE_OPTIONS_SHOW, {
      options: availableOptions,
    });
  }

  private _endDialogue(): void {
    const dialogueId = this._state.currentDialogueId;
    
    // 处理结束回调
    if (this._currentDialogue?.onEnd) {
      this._handleCallback(this._currentDialogue.onEnd);
    }

    this._currentDialogue = null;
    this._state.currentDialogueId = null;

    eventBus.emit(GameEvent.DIALOGUE_END, { dialogueId });
    this._onEndCallback?.();
  }

  private _applyEffects(effects: IDialogueEffects): void {
    // R值变化
    if (effects.rChange) {
      worldState.addR(effects.rChange, `dialogue_choice`);
    }

    // P值变化
    if (effects.pChange) {
      worldState.addP(effects.pChange, `dialogue_choice`);
    }

    // 设置标记
    if (effects.setFlags) {
      worldState.setFlags(effects.setFlags);
    }

    // 获得卡片
    if (effects.obtainCard) {
      eventBus.emit(GameEvent.CARD_OBTAIN_REQUEST, { cardId: effects.obtainCard });
    }

    // 触发伏笔
    if (effects.triggerForeshadow) {
      eventBus.emit(GameEvent.FORESHADOW_TRIGGER_REQUEST, effects.triggerForeshadow);
    }

    // 播放音效
    if (effects.playSfx) {
      eventBus.emit(GameEvent.SFX_PLAY, { key: effects.playSfx });
    }

    // 自定义回调
    if (effects.callback) {
      this._handleCallback(effects.callback);
    }
  }

  private _handleCallback(callback: { type: string; name: string; params?: Record<string, unknown> }): void {
    if (callback.type === 'event') {
      eventBus.emit(callback.name as any, callback.params);
    }
    // function类型在外部注册处理
  }

  private _recordHistory(speaker: string, text: string, selectedOption?: string): void {
    this._state.history.push({
      dialogueId: this._state.currentDialogueId!,
      speaker,
      text,
      timestamp: Date.now(),
      selectedOption,
    });

    // 限制历史长度
    if (this._state.history.length > this._config.historyLimit) {
      this._state.history.shift();
    }
  }

  private _wait(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
  }

  private _createInitialState(): IDialogueState {
    return {
      currentDialogueId: null,
      currentLineIndex: 0,
      isTyping: false,
      isWaitingInput: false,
      optionsVisible: false,
      history: [],
    };
  }

  /**
   * 序列化历史
   */
  serializeHistory(): IDialogueState['history'] {
    return [...this._state.history];
  }
}
```

### 3.1 CardSystem实现

```typescript
// src/systems/narrative/CardSystem.ts

import { 
  ICard, 
  CardCategory, 
  IObtainedCard, 
  ICardCollection,
  ICardSystemConfig,
  IVerdictCardParams,
  TZoneId 
} from '@/types';
import { eventBus, GameEvent } from '@/systems/EventBus';
import { worldState } from '@/systems/world';

/**
 * 默认配置
 */
const DEFAULT_CONFIG: ICardSystemConfig = {
  newCardToastDuration: 3000,
  recentLimit: 5,
  autoMarkRead: false,
};

/**
 * 卡片系统
 */
export class CardSystem {
  private _cards: Map<string, ICard> = new Map();
  private _collection: ICardCollection;
  private _config: ICardSystemConfig;
  private _verdictCounter: number = 0;

  constructor(config: Partial<ICardSystemConfig> = {}) {
    this._config = { ...DEFAULT_CONFIG, ...config };
    this._collection = this._createEmptyCollection();
  }

  /**
   * 注册卡片数据
   */
  registerCards(cards: Record<string, ICard>): void {
    for (const [id, card] of Object.entries(cards)) {
      this._cards.set(id, card);
    }
  }

  /**
   * 获取卡片定义
   */
  getCard(cardId: string): ICard | undefined {
    return this._cards.get(cardId);
  }

  /**
   * 获得卡片
   */
  obtain(cardId: string, zoneId?: TZoneId): boolean {
    // 检查卡片是否存在
    const card = this.getCard(cardId);
    if (!card) {
      console.warn(`Card ${cardId} not found`);
      return false;
    }

    // 检查是否已获得
    if (this.hasObtained(cardId)) {
      return false;
    }

    // 创建获得记录
    const obtainedCard: IObtainedCard = {
      cardId,
      obtainedAt: Date.now(),
      obtainedZone: zoneId || worldState.getCurrentZone() || ('C0-Z1' as TZoneId),
      isRead: false,
      readCount: 0,
    };

    // 添加到收藏
    this._collection.obtained.push(obtainedCard);
    this._collection.countByCategory[card.category]++;
    this._collection.totalCount++;
    
    // 更新最近获得
    this._collection.recentlyObtained.unshift(cardId);
    if (this._collection.recentlyObtained.length > this._config.recentLimit) {
      this._collection.recentlyObtained.pop();
    }

    // 设置标记
    worldState.setFlag(`card_${cardId}_obtained`, true);

    // 记录到Zone
    if (zoneId) {
      worldState.getZoneState(zoneId).obtainedCards.push(cardId);
    }

    // 发送事件
    eventBus.emit(GameEvent.CARD_OBTAIN, {
      cardId,
      cardType: card.category,
      card,
    });

    return true;
  }

  /**
   * 检查是否已获得
   */
  hasObtained(cardId: string): boolean {
    return this._collection.obtained.some(c => c.cardId === cardId);
  }

  /**
   * 获取已获得卡片
   */
  getObtainedCard(cardId: string): IObtainedCard | undefined {
    return this._collection.obtained.find(c => c.cardId === cardId);
  }

  /**
   * 标记已读
   */
  markRead(cardId: string): void {
    const obtained = this.getObtainedCard(cardId);
    if (obtained) {
      obtained.isRead = true;
      obtained.readCount++;
    }
  }

  /**
   * 获取收藏状态
   */
  getCollection(): ICardCollection {
    return { ...this._collection };
  }

  /**
   * 按类别获取已获得卡片
   */
  getByCategory(category: CardCategory): ICard[] {
    return this._collection.obtained
      .filter(o => {
        const card = this.getCard(o.cardId);
        return card?.category === category;
      })
      .map(o => this.getCard(o.cardId)!)
      .filter(Boolean);
  }

  /**
   * 获取未读卡片数量
   */
  getUnreadCount(): number {
    return this._collection.obtained.filter(c => !c.isRead).length;
  }

  /**
   * 生成判定卡
   */
  generateVerdict(params: IVerdictCardParams): ICard {
    const id = `VERDICT_${++this._verdictCounter}`;
    
    const card: ICard = {
      id,
      category: CardCategory.VERDICT,
      title: '系统判定',
      subtitle: params.trigger,
      content: params.verdict,
      obtainedAt: params.zone,
      meta: {
        counters: params.counters,
        generatedAt: Date.now(),
      },
    };

    // 注册并获得
    this._cards.set(id, card);
    this.obtain(id, params.zone);

    return card;
  }

  /**
   * 检查收集进度
   */
  getProgress(): { obtained: number; total: number; percentage: number } {
    const total = this._cards.size;
    const obtained = this._collection.totalCount;
    const percentage = total > 0 ? Math.round((obtained / total) * 100) : 0;
    
    return { obtained, total, percentage };
  }

  /**
   * 按伏笔获取关联卡片
   */
  getByForeshadow(foreshadowId: string): ICard[] {
    return Array.from(this._cards.values())
      .filter(card => card.relatedForeshadow === foreshadowId);
  }

  /**
   * 按角色获取关联卡片
   */
  getByCharacter(characterId: string): ICard[] {
    return Array.from(this._cards.values())
      .filter(card => card.relatedCharacter === characterId);
  }

  /**
   * 序列化
   */
  serialize(): { obtained: IObtainedCard[]; verdictCounter: number } {
    return {
      obtained: [...this._collection.obtained],
      verdictCounter: this._verdictCounter,
    };
  }

  /**
   * 反序列化
   */
  deserialize(data: { obtained: IObtainedCard[]; verdictCounter: number }): void {
    this._collection = this._createEmptyCollection();
    
    for (const obtained of data.obtained) {
      this._collection.obtained.push(obtained);
      const card = this.getCard(obtained.cardId);
      if (card) {
        this._collection.countByCategory[card.category]++;
        this._collection.totalCount++;
      }
    }

    this._verdictCounter = data.verdictCounter;
  }

  /**
   * 重置
   */
  reset(): void {
    this._collection = this._createEmptyCollection();
    this._verdictCounter = 0;
  }

  private _createEmptyCollection(): ICardCollection {
    return {
      obtained: [],
      countByCategory: {
        [CardCategory.ARCHIVE]: 0,
        [CardCategory.ITEM]: 0,
        [CardCategory.PRAYER]: 0,
        [CardCategory.VERDICT]: 0,
      },
      totalCount: 0,
      recentlyObtained: [],
    };
  }
}
```

## 4. 伏笔追踪系统

### 4.1 ForeshadowTracker实现

```typescript
// src/systems/narrative/ForeshadowTracker.ts

import { IForeshadow, ForeshadowState, TZoneId } from '@/types';
import { eventBus, GameEvent } from '@/systems/EventBus';
import { worldState } from '@/systems/world';

/**
 * 伏笔追踪器
 */
export class ForeshadowTracker {
  private _foreshadows: Map<string, IForeshadow> = new Map();

  /**
   * 注册伏笔定义
   */
  registerForeshadows(foreshadows: IForeshadow[]): void {
    for (const f of foreshadows) {
      this._foreshadows.set(f.id, { ...f, state: ForeshadowState.HIDDEN });
    }
  }

  /**
   * 获取伏笔
   */
  get(foreshadowId: string): IForeshadow | undefined {
    return this._foreshadows.get(foreshadowId);
  }

  /**
   * 获取伏笔状态
   */
  getState(foreshadowId: string): ForeshadowState {
    return this._foreshadows.get(foreshadowId)?.state || ForeshadowState.HIDDEN;
  }

  /**
   * 投放伏笔
   */
  plant(foreshadowId: string, zoneId?: TZoneId): boolean {
    const foreshadow = this._foreshadows.get(foreshadowId);
    if (!foreshadow) return false;

    // 只能从HIDDEN状态投放
    if (foreshadow.state !== ForeshadowState.HIDDEN) {
      return false;
    }

    const zone = zoneId || worldState.getCurrentZone();
    
    foreshadow.state = ForeshadowState.PLANTED;
    foreshadow.plantedAt = zone || undefined;
    foreshadow.plantedTime = Date.now();

    // 设置标记
    worldState.setFlag(`foreshadow_${foreshadowId}_planted`, true);

    // 记录到Zone
    if (zone) {
      worldState.getZoneState(zone).triggeredForeshadows.push(foreshadowId);
    }

    eventBus.emit(GameEvent.FORESHADOW_PLANT, {
      foreshadowId,
      zoneId: zone,
      foreshadow,
    });

    return true;
  }

  /**
   * 加深伏笔
   */
  deepen(foreshadowId: string, zoneId?: TZoneId): boolean {
    const foreshadow = this._foreshadows.get(foreshadowId);
    if (!foreshadow) return false;

    // 只能从PLANTED状态加深（可多次加深）
    if (foreshadow.state !== ForeshadowState.PLANTED && 
        foreshadow.state !== ForeshadowState.DEEPENED) {
      return false;
    }

    const zone = zoneId || worldState.getCurrentZone();
    
    foreshadow.state = ForeshadowState.DEEPENED;
    
    if (!foreshadow.deepenedAt) {
      foreshadow.deepenedAt = [];
    }
    
    foreshadow.deepenedAt.push({
      zone: zone!,
      timestamp: Date.now(),
    });

    // 设置标记
    worldState.setFlag(`foreshadow_${foreshadowId}_deepened`, true);

    eventBus.emit(GameEvent.FORESHADOW_DEEPEN, {
      foreshadowId,
      zoneId: zone,
      deepenCount: foreshadow.deepenedAt.length,
    });

    return true;
  }

  /**
   * 回收伏笔
   */
  recall(foreshadowId: string, zoneId?: TZoneId): boolean {
    const foreshadow = this._foreshadows.get(foreshadowId);
    if (!foreshadow) return false;

    // 只能从PLANTED或DEEPENED状态回收
    if (foreshadow.state !== ForeshadowState.PLANTED && 
        foreshadow.state !== ForeshadowState.DEEPENED) {
      return false;
    }

    const zone = zoneId || worldState.getCurrentZone();
    
    foreshadow.state = ForeshadowState.RECALLED;
    foreshadow.recalledAt = zone || undefined;
    foreshadow.recalledTime = Date.now();

    // 设置标记
    worldState.setFlag(`foreshadow_${foreshadowId}_recalled`, true);

    eventBus.emit(GameEvent.FORESHADOW_RECALL, {
      foreshadowId,
      zoneId: zone,
      foreshadow,
    });

    return true;
  }

  /**
   * 触发伏笔（通用入口）
   */
  trigger(foreshadowId: string, action: 'plant' | 'deepen' | 'recall', zoneId?: TZoneId): boolean {
    switch (action) {
      case 'plant': return this.plant(foreshadowId, zoneId);
      case 'deepen': return this.deepen(foreshadowId, zoneId);
      case 'recall': return this.recall(foreshadowId, zoneId);
      default: return false;
    }
  }

  /**
   * 检查伏笔是否已投放
   */
  isPlanted(foreshadowId: string): boolean {
    const state = this.getState(foreshadowId);
    return state === ForeshadowState.PLANTED || 
           state === ForeshadowState.DEEPENED || 
           state === ForeshadowState.RECALLED;
  }

  /**
   * 检查伏笔是否已回收
   */
  isRecalled(foreshadowId: string): boolean {
    return this.getState(foreshadowId) === ForeshadowState.RECALLED;
  }

  /**
   * 获取所有伏笔
   */
  getAll(): IForeshadow[] {
    return Array.from(this._foreshadows.values());
  }

  /**
   * 按状态获取伏笔
   */
  getByState(state: ForeshadowState): IForeshadow[] {
    return this.getAll().filter(f => f.state === state);
  }

  /**
   * 获取进度
   */
  getProgress(): { 
    total: number; 
    planted: number; 
    deepened: number; 
    recalled: number; 
    hidden: number;
  } {
    const all = this.getAll();
    return {
      total: all.length,
      planted: all.filter(f => f.state === ForeshadowState.PLANTED).length,
      deepened: all.filter(f => f.state === ForeshadowState.DEEPENED).length,
      recalled: all.filter(f => f.state === ForeshadowState.RECALLED).length,
      hidden: all.filter(f => f.state === ForeshadowState.HIDDEN).length,
    };
  }

  /**
   * 检查Zone相关伏笔
   */
  getZoneForeshadows(zoneId: TZoneId): {
    planted: IForeshadow[];
    deepened: IForeshadow[];
    recalled: IForeshadow[];
  } {
    const result = {
      planted: [] as IForeshadow[],
      deepened: [] as IForeshadow[],
      recalled: [] as IForeshadow[],
    };

    for (const f of this._foreshadows.values()) {
      if (f.plantedAt === zoneId) {
        result.planted.push(f);
      }
      if (f.deepenedAt?.some(d => d.zone === zoneId)) {
        result.deepened.push(f);
      }
      if (f.recalledAt === zoneId) {
        result.recalled.push(f);
      }
    }

    return result;
  }

  /**
   * 序列化
   */
  serialize(): Record<string, IForeshadow> {
    const result: Record<string, IForeshadow> = {};
    for (const [id, f] of this._foreshadows) {
      result[id] = { ...f };
    }
    return result;
  }

  /**
   * 反序列化
   */
  deserialize(data: Record<string, IForeshadow>): void {
    for (const [id, f] of Object.entries(data)) {
      if (this._foreshadows.has(id)) {
        const existing = this._foreshadows.get(id)!;
        Object.assign(existing, f);
      }
    }
  }

  /**
   * 重置
   */
  reset(): void {
    for (const f of this._foreshadows.values()) {
      f.state = ForeshadowState.HIDDEN;
      f.plantedAt = undefined;
      f.plantedTime = undefined;
      f.deepenedAt = undefined;
      f.recalledAt = undefined;
      f.recalledTime = undefined;
    }
  }
}
```

```typescript
  setDialogueCallbacks(callbacks: Parameters<DialogueManager['setCallbacks']>[0]): void {
```
