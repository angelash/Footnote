## 1. 架构概述

### 1.1 技术栈选型

| 层级 | 技术 | 版本 | 选型理由 |
|------|------|------|----------|
| **游戏引擎** | Phaser 3 | 3.70+ | H5游戏首选，Canvas/WebGL双渲染，丰富的2D功能 |
| **开发语言** | TypeScript | 5.0+ | 类型安全，IDE支持好，重构友好 |
| **构建工具** | Vite | 5.0+ | 快速HMR，ES模块原生支持 |
| **本地存储** | IndexedDB | - | 大容量，异步，结构化存储 |
| **单元测试** | Vitest | 1.0+ | Vite生态，快速，TS原生支持 |
| **E2E测试** | Playwright | 1.40+ | 跨浏览器，可靠，支持移动端模拟 |
| **数据格式** | YAML | - | 可读性强，适合叙事内容配置 |

### 1.3 模块职责划分

| 模块 | 职责 | 依赖 | 被依赖 |
|------|------|------|--------|
| **BootScene** | 引擎初始化、配置加载 | Phaser | PreloadScene |
| **PreloadScene** | 资源预加载、进度显示 | AssetRegistry | MenuScene/GameScene |
| **GameScene** | 游戏主场景、实体管理 | 所有Systems | UI Layer |
| **NarrativeEngine** | 对话、卡片、伏笔管理 | WorldState | GameScene |
| **WorldState** | 计数器、状态、标记管理 | EventBus | NarrativeEngine、SaveManager |
| **AbilitySystem** | 三种能力的激活与效果 | WorldState | GameScene |
| **ZoneManager** | Zone切换、资源协调 | ConfigLoader | GameScene |
| **SaveManager** | 存档读写、自动保存 | StorageAdapter | 所有Systems |
| **AudioManager** | BGM/SFX/Ambience控制 | AssetRegistry | 所有Scenes |
| **EventBus** | 系统间解耦通信 | 无 | 所有Systems |

// 事件负载类型定义
export interface IEventPayloads {
  [GameEvent.ZONE_ENTER]: { zoneId: string; isRevisit: boolean };
  [GameEvent.ZONE_EXIT]: { zoneId: string; nextZoneId: string };
  [GameEvent.PLAYER_MOVE]: { x: number; y: number; direction: string };
  [GameEvent.COUNTER_CHANGE]: { type: 'R' | 'P' | 'W'; oldValue: number; newValue: number };
  [GameEvent.THRESHOLD_REACHED]: { type: 'R' | 'P' | 'W'; level: string; value: number };
  [GameEvent.DIALOGUE_START]: { dialogueId: string; speaker?: string };
  [GameEvent.DIALOGUE_CHOICE]: { choiceId: string; dialogueId: string };
  [GameEvent.CARD_OBTAIN]: { cardId: string; cardType: string };
  [GameEvent.FORESHADOW_PLANT]: { foreshadowId: string; zoneId: string };
  [GameEvent.ABILITY_ACTIVATE]: { abilityId: string; targetId?: string };
  [GameEvent.TIME_NODE_CREATED]: { nodeId: string; timestamp: number; zoneId: string };
  // ... 其他事件负载
}

// 全局事件总线实例
class GameEventBus extends Phaser.Events.EventEmitter {
  private static _instance: GameEventBus;
  
  private constructor() {
    super();
  }
  
  static getInstance(): GameEventBus {
    if (!GameEventBus._instance) {
      GameEventBus._instance = new GameEventBus();
    }
    return GameEventBus._instance;
  }
  
  // 类型安全的emit
  emitEvent<K extends keyof IEventPayloads>(
    event: K,
    payload: IEventPayloads[K]
  ): boolean {
    return this.emit(event, payload);
  }
  
  // 类型安全的on
  onEvent<K extends keyof IEventPayloads>(
    event: K,
    fn: (payload: IEventPayloads[K]) => void,
    context?: unknown
  ): this {
    return this.on(event, fn, context);
  }
}

export const eventBus = GameEventBus.getInstance();

  private _services: Map<string, unknown> = new Map();
  private _initialized: boolean = false;
  
  // 注册服务
  register<T>(key: string, factory: () => T): void {
    if (this._services.has(key)) {
      console.warn(`Service ${key} already registered, overwriting.`);
    }
    this._services.set(key, factory);
  }
  
  // 获取服务（懒加载）
  get<T>(key: string): T {
    const factoryOrInstance = this._services.get(key);
    if (!factoryOrInstance) {
      throw new Error(`Service ${key} not found`);
    }
    
    // 如果是工厂函数，执行并替换为实例
    if (typeof factoryOrInstance === 'function') {
      const instance = (factoryOrInstance as () => T)();
      this._services.set(key, instance);
      return instance;
    }
    
    return factoryOrInstance as T;
  }
  
  // 便捷访问器
  get worldState(): WorldState {
    return this.get<WorldState>('worldState');
  }
  
  get narrativeEngine(): NarrativeEngine {
    return this.get<NarrativeEngine>('narrativeEngine');
  }
  
  get abilitySystem(): AbilitySystem {
    return this.get<AbilitySystem>('abilitySystem');
  }
  
  get saveManager(): SaveManager {
    return this.get<SaveManager>('saveManager');
  }
  
  get audioManager(): AudioManager {
    return this.get<AudioManager>('audioManager');
  }
  
  get zoneManager(): ZoneManager {
    return this.get<ZoneManager>('zoneManager');
  }
  
  get inputManager(): InputManager {
    return this.get<InputManager>('inputManager');
  }
  
  get uiManager(): UIManager {
    return this.get<UIManager>('uiManager');
  }
  
  // 初始化所有服务
  async initialize(scene: Phaser.Scene): Promise<void> {
    if (this._initialized) return;
    
    // 按依赖顺序初始化
    this.register('worldState', () => new WorldState());
    this.register('saveManager', () => new SaveManager(new IndexedDBAdapter()));
    this.register('audioManager', () => new AudioManager(scene));
    this.register('narrativeEngine', () => new NarrativeEngine(this.worldState));
    this.register('abilitySystem', () => new AbilitySystem(this.worldState));
    this.register('zoneManager', () => new ZoneManager(scene));
    this.register('inputManager', () => new InputManager(scene));
    this.register('uiManager', () => new UIManager(scene));
    
    // 初始化存储
    await this.saveManager.initialize();
    
    this._initialized = true;
  }
}

export const services = new ServiceContainer();

interface IStateChange<T> {
  path: string;
  oldValue: T;
  newValue: T;
  timestamp: number;
  source: string;
}

// 状态订阅机制
class StateObservable<T> {
  private _value: T;
  private _observers: Set<(newValue: T, oldValue: T) => void> = new Set();
  
  constructor(initialValue: T) {
    this._value = initialValue;
  }
  
  get value(): T {
    return this._value;
  }
  
  set(newValue: T): void {
    const oldValue = this._value;
    if (oldValue !== newValue) {
      this._value = newValue;
      this._observers.forEach(fn => fn(newValue, oldValue));
    }
  }
  
  subscribe(fn: (newValue: T, oldValue: T) => void): () => void {
    this._observers.add(fn);
    return () => this._observers.delete(fn);
  }
}

// src/utils/ErrorHandler.ts

enum ErrorSeverity {
  FATAL = 'fatal',
  ERROR = 'error',
  WARNING = 'warning',
  INFO = 'info',
}

interface IGameError {
  code: string;
  message: string;
  severity: ErrorSeverity;
  context?: Record<string, unknown>;
  stack?: string;
}

class ErrorHandler {
  private _errors: IGameError[] = [];
  private _maxErrors: number = 100;
  
  // 处理错误
  handle(error: Error | IGameError, severity: ErrorSeverity = ErrorSeverity.ERROR): void {
    const gameError: IGameError = error instanceof Error
      ? this._convertError(error, severity)
      : error;
    
    this._errors.push(gameError);
    if (this._errors.length > this._maxErrors) {
      this._errors.shift();
    }
    
    // 根据严重程度处理
    switch (severity) {
      case ErrorSeverity.FATAL:
        this._handleFatal(gameError);
        break;
      case ErrorSeverity.ERROR:
        this._handleError(gameError);
        break;
      case ErrorSeverity.WARNING:
        console.warn(`[WARNING] ${gameError.code}: ${gameError.message}`);
        break;
      case ErrorSeverity.INFO:
        console.info(`[INFO] ${gameError.code}: ${gameError.message}`);
        break;
    }
  }
  
  private _convertError(error: Error, severity: ErrorSeverity): IGameError {
    return {
      code: 'RUNTIME_ERROR',
      message: error.message,
      severity,
      stack: error.stack,
    };
  }
  
  private _handleFatal(error: IGameError): void {
    console.error(`[FATAL] ${error.code}: ${error.message}`, error.stack);
    eventBus.emit(GameEvent.GAME_FATAL_ERROR, error);
    // 显示错误UI
  }
  
  private _handleError(error: IGameError): void {
    console.error(`[ERROR] ${error.code}: ${error.message}`, error.context);
    // 尝试恢复或记录
  }
  
  // 获取错误日志（用于导出/上报）
  getErrorLog(): IGameError[] {
    return [...this._errors];
  }
}

export const errorHandler = new ErrorHandler();

// 全局错误捕获
window.addEventListener('error', (event) => {
  errorHandler.handle(event.error, ErrorSeverity.ERROR);
});

window.addEventListener('unhandledrejection', (event) => {
  errorHandler.handle(new Error(String(event.reason)), ErrorSeverity.ERROR);
});

interface IResourceConfig {
  critical: string[];      // 必须加载

class ResourceManager {
  private _loadedResources: Set<string> = new Set();
  private _loadingPromises: Map<string, Promise<void>> = new Map();
  private _scene: Phaser.Scene;
  
  constructor(scene: Phaser.Scene) {
    this._scene = scene;
  }
  
  // 按Zone预加载资源
  async preloadForZone(zoneId: string): Promise<void> {
    const config = await this._getZoneResourceConfig(zoneId);
    
    // 加载当前Zone资源
    await this._loadResources(config.critical);
    
    // 后台预加载相邻Zone
    const adjacentZones = this._getAdjacentZones(zoneId);
    this._backgroundLoad(adjacentZones);
  }
  
  // 卸载不需要的资源
  unloadUnused(currentZoneId: string, adjacentZoneIds: string[]): void {
    const neededKeys = this._getNeededResourceKeys(currentZoneId, adjacentZoneIds);
    
    this._loadedResources.forEach(key => {
      if (!neededKeys.has(key) && !this._isCritical(key)) {
        this._unloadResource(key);
      }
    });
  }
  
  private async _loadResources(keys: string[]): Promise<void> {
    const toLoad = keys.filter(key => !this._loadedResources.has(key));
    
    if (toLoad.length === 0) return;
    
    return new Promise((resolve) => {
      toLoad.forEach(key => {
        // 根据key类型加载对应资源
        this._loadByType(key);
      });
      
      this._scene.load.once('complete', () => {
        toLoad.forEach(key => this._loadedResources.add(key));
        resolve();
      });
      
      this._scene.load.start();
    });
  }
  
  private _backgroundLoad(zoneIds: string[]): void {
    // 使用requestIdleCallback在空闲时加载
    requestIdleCallback(async () => {
      for (const zoneId of zoneIds) {
        const config = await this._getZoneResourceConfig(zoneId);
        await this._loadResources(config.preload);
      }
    });
  }
}

    shadowEnabled: true,
    postProcessing: true,

    shadowEnabled: false,
    postProcessing: false,
  },

    targetFPS: 30,
    renderScale: 0.6,
    particleLimit: 50,
    shadowEnabled: false,
    postProcessing: false,
  },
};

class PerformanceManager {
  private _currentProfile: IPerformanceProfile;
  private _fpsHistory: number[] = [];
  private _autoAdjust: boolean = true;
  
  constructor() {
    this._currentProfile = PERFORMANCE_PROFILES.high;
  }
  
  // 自动调整性能配置
  update(deltaTime: number): void {
    if (!this._autoAdjust) return;
    
    const fps = 1000 / deltaTime;
    this._fpsHistory.push(fps);
    
    if (this._fpsHistory.length >= 60) {
      const avgFPS = this._fpsHistory.reduce((a, b) => a + b) / this._fpsHistory.length;
      this._adjustProfile(avgFPS);
      this._fpsHistory = [];
    }
  }
  
  private _adjustProfile(avgFPS: number): void {
    if (avgFPS < 25 && this._currentProfile !== PERFORMANCE_PROFILES.low) {
      this._applyProfile(PERFORMANCE_PROFILES.low);
    } else if (avgFPS < 45 && this._currentProfile === PERFORMANCE_PROFILES.high) {
      this._applyProfile(PERFORMANCE_PROFILES.medium);
    } else if (avgFPS > 55 && this._currentProfile === PERFORMANCE_PROFILES.low) {
      this._applyProfile(PERFORMANCE_PROFILES.medium);
    }
  }
  
  private _applyProfile(profile: IPerformanceProfile): void {
    this._currentProfile = profile;
    eventBus.emit('performance:profile:change', profile);
  }
}

// src/systems/save/SaveIntegrity.ts

import { createHash } from 'crypto';

class SaveIntegrity {
  private _secret: string = '__FOOTNOTE_GAME__';
  
  // 生成签名
  sign(data: ISaveData): string {
    const content = JSON.stringify(data);
    return this._hash(content + this._secret);
  }
  
  // 验证签名
  verify(data: ISaveData, signature: string): boolean {
    const expectedSignature = this.sign(data);
    return expectedSignature === signature;
  }
  
  // 简单哈希（浏览器环境）
  private _hash(input: string): string {
    let hash = 0;
    for (let i = 0; i < input.length; i++) {
      const char = input.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash;
    }
    return hash.toString(16);
  }
}
