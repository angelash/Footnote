# 《备注 / Footnote》数据结构与接口设计文档

> **文档版本**：v1.0  
> **文档性质**：接口规范  
> **适用对象**：全栈开发  
> **最后更新**：2025-12-24

## 1. 类型系统概述

### 1.1 类型文件组织

```
src/types/
├── index.ts          # 统一导出
├── common.ts         # 通用类型
├── scene.ts          # 场景相关类型
├── dialogue.ts       # 对话相关类型
├── card.ts           # 卡片相关类型
├── world.ts          # 世界状态类型
├── ability.ts        # 能力系统类型
├── save.ts           # 存档相关类型
├── audio.ts          # 音频相关类型
├── ui.ts             # UI相关类型
└── events.ts         # 事件负载类型
```

### 1.2 命名约定

| 类型 | 前缀 | 示例 |
|------|------|------|
| **接口** | `I` | `IDialogue`, `ICard` |
| **类型别名** | `T` | `TZoneId`, `TCardType` |
| **枚举** | 无 | `AbilityType`, `CardCategory` |
| **泛型参数** | `T`, `K`, `V` | `IRepository<T>` |

// src/types/common.ts

/**
 * 二维坐标
 */
export interface IPosition {
  x: number;
  y: number;
}

/**
 * 尺寸
 */
export interface ISize {
  width: number;
  height: number;
}

/**
 * 矩形区域
 */
export interface IRect extends IPosition, ISize {}

/**
 * 边界框
 */
export interface IBounds {
  left: number;
  top: number;
  right: number;
  bottom: number;
}

/**
 * 颜色值
 */
export type TColor = number | string;

/**
 * 时间戳（毫秒）
 */
export type TTimestamp = number;

/**
 * 唯一标识符
 */
export type TUniqueId = string;

/**
 * Zone ID 格式: C{chapter}-Z{zone}
 * 示例: C0-Z1, C3-Z5, CF-Z2
 */
export type TZoneId = `C${number | 'F'}-Z${number}`;

/**
 * 章节ID
 */
export type TChapterId = 'C0' | 'C1' | 'C2' | 'C3' | 'C4' | 'C5' | 'CF';

/**
 * 方向
 */
export enum Direction {
  UP = 'up',
  DOWN = 'down',
  LEFT = 'left',
  RIGHT = 'right',
  UP_LEFT = 'up_left',
  UP_RIGHT = 'up_right',
  DOWN_LEFT = 'down_left',
  DOWN_RIGHT = 'down_right',
}

/**
 * 可序列化接口
 */
export interface ISerializable<T> {
  serialize(): T;
  deserialize(data: T): void;
}

/**
 * 可销毁接口
 */
export interface IDestroyable {
  destroy(): void;
}

/**
 * 可更新接口
 */
export interface IUpdatable {
  update(delta: number): void;
}

/**
 * 仓库接口
 */
export interface IRepository<T, K = string> {
  get(id: K): T | undefined;
  getAll(): T[];
  has(id: K): boolean;
  add(item: T): void;
  remove(id: K): boolean;
  clear(): void;
}

/**
 * 状态变更记录
 */
export interface IStateChange<T = unknown> {
  path: string;
  oldValue: T;
  newValue: T;
  timestamp: TTimestamp;
  source: string;
}

/**
 * 异步操作结果
 */
export type TResult<T, E = Error> = 
  | { success: true; data: T }
  | { success: false; error: E };

/**
 * 可选深度
 */
export type DeepPartial<T> = {
  [P in keyof T]?: T[P] extends object ? DeepPartial<T[P]> : T[P];
};

import { ISceneConfig, IDialogue, ICard, ISaveData } from '@/types';

/**
 * 检查是否为有效的场景配置
 */
export function isSceneConfig(data: unknown): data is ISceneConfig {
  if (typeof data !== 'object' || data === null) return false;
  const config = data as Partial<ISceneConfig>;
  return (
    typeof config.id === 'string' &&
    typeof config.title === 'string' &&
    typeof config.background === 'object' &&
    Array.isArray(config.objects)
  );
}

/**
 * 检查是否为有效的对话数据
 */
export function isDialogue(data: unknown): data is IDialogue {
  if (typeof data !== 'object' || data === null) return false;
  const dialogue = data as Partial<IDialogue>;
  return (
    typeof dialogue.id === 'string' &&
    typeof dialogue.speaker === 'string' &&
    Array.isArray(dialogue.lines)
  );
}

/**
 * 检查是否为有效的卡片数据
 */
export function isCard(data: unknown): data is ICard {
  if (typeof data !== 'object' || data === null) return false;
  const card = data as Partial<ICard>;
  return (
    typeof card.id === 'string' &&
    typeof card.category === 'string' &&
    typeof card.title === 'string' &&
    typeof card.content === 'string'
  );
}

/**
 * 检查是否为有效的存档数据
 */
export function isSaveData(data: unknown): data is ISaveData {
  if (typeof data !== 'object' || data === null) return false;
  const save = data as Partial<ISaveData>;
  return (
    typeof save.meta === 'object' &&
    typeof save.world === 'object' &&
    typeof save.player === 'object' &&
    typeof save.collection === 'object' &&
    Array.isArray(save.timeNodes) &&
    typeof save.settings === 'object'
  );
}
