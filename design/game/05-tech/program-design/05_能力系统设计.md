## 1. 系统概述

### 1.1 三种核心能力

| 能力 | 解锁章节 | 操作方式 | 核心体验 | 代价 |
|------|----------|----------|----------|------|
| **深度感知** | C2-Z1 | 长按屏幕 | "看见而不能改变" | P轻微增加 |
| **深度介入** | C3-Z1 | 拖拽操作 | "改变留下伤痕" | 永久伤痕 + P增加 |
| **时间干预** | C4-Z2 | 时间线UI | "回溯洗不净" | 污染 + P大增 |

## 2. 深度感知 (Depth Perception)

### 2.1 机制设计

```
触发流程:
┌────────────┐
│ 触摸开始   │
└─────┬──────┘
      │

┌────────────┐     0.5秒

└─────┬──────┘
      │ 长按超过0.5秒

┌────────────┐

└─────┬──────┘

└─────┬──────┘
```

### 2.2 DepthPerception实现

```typescript
// src/systems/ability/DepthPerception.ts

import { IPerceptionState, IPosition } from '@/types';
import { eventBus, GameEvent } from '@/systems/EventBus';
import { worldState } from '@/systems/world';

/**
 * 深度感知配置
 */
interface IDepthPerceptionConfig {
  /** 激活所需长按时间 (ms) */
  chargeTime: number;
  /** P值计算间隔 (s) */
  pIntervalSeconds: number;
  /** P值增长表 */
  pCostByDuration: Array<{ maxSeconds: number; pCost: number }>;
  /** 视觉效果配置 */
  effects: {
    vignetteIntensity: number;
    scanlineAlpha: number;
    colorShift: number;
  };
}

const DEFAULT_CONFIG: IDepthPerceptionConfig = {
  chargeTime: 500,
  pIntervalSeconds: 5,
  pCostByDuration: [
    { maxSeconds: 5, pCost: 0 },
    { maxSeconds: 15, pCost: 0.1 },
    { maxSeconds: 30, pCost: 0.2 },
    { maxSeconds: Infinity, pCost: 0.3 },
  ],
  effects: {
    vignetteIntensity: 0.4,
    scanlineAlpha: 0.1,
    colorShift: -20,
  },
};

/**
 * 深度感知能力
 */
export class DepthPerception {
  private _config: IDepthPerceptionConfig;
  private _state: IPerceptionState;
  private _chargeTimer: number | null = null;
  private _activeTimer: number | null = null;
  private _pAccumulateTimer: number | null = null;
  private _isUnlocked: boolean = false;
  private _revealableObjects: Map<string, IRevealableObject> = new Map();

  constructor(config: Partial<IDepthPerceptionConfig> = {}) {
    this._config = { ...DEFAULT_CONFIG, ...config };
    this._state = this._createInitialState();
  }

  /**
   * 解锁能力
   */
  unlock(): void {
    this._isUnlocked = true;
    worldState.setFlag('ability_perception_unlocked', true);
    
    eventBus.emit(GameEvent.ABILITY_UNLOCK, {
      abilityId: 'depth_perception',
    });
  }

  /**
   * 是否已解锁
   */
  isUnlocked(): boolean {
    return this._isUnlocked;
  }

  /**
   * 开始蓄力（触摸开始）
   */
  startCharge(): void {
    if (!this._isUnlocked || this._state.isActive) return;

    eventBus.emit(GameEvent.GESTURE_LONG_PRESS_START);

    // 开始蓄力计时
    this._chargeTimer = window.setTimeout(() => {
      this._activate();
    }, this._config.chargeTime);
  }

  /**
   * 取消蓄力（触摸结束但未激活）
   */
  cancelCharge(): void {
    if (this._chargeTimer) {
      clearTimeout(this._chargeTimer);
      this._chargeTimer = null;
    }
  }

  /**
   * 激活感知
   */
  private _activate(): void {
    this._chargeTimer = null;
    this._state.isActive = true;
    this._state.startTime = Date.now();

    // 播放激活音效
    eventBus.emit(GameEvent.SFX_PLAY, { key: 'sfx_perception_activate' });

    // 发送激活事件
    eventBus.emit(GameEvent.PERCEPTION_START, {
      startTime: this._state.startTime,
    });

    eventBus.emit(GameEvent.ABILITY_ACTIVATE, {
      abilityId: 'depth_perception',
    });

    // 显示隐藏对象
    this._revealObjects();

    // 开始P值累积计时
    this._startPAccumulation();
  }

  /**
   * 停用感知（松开触摸）
   */
  deactivate(): void {
    if (!this._state.isActive) {
      this.cancelCharge();
      return;
    }

    // 计算总持续时间
    const duration = Date.now() - (this._state.startTime || 0);
    this._state.totalDuration += duration;

    // 计算最终P值消耗
    const pCost = this._calculatePCost(duration);
    if (pCost > 0) {
      worldState.addP(pCost, 'depth_perception');
    }

    // 停止P值累积
    this._stopPAccumulation();

    // 重置状态
    this._state.isActive = false;
    this._state.startTime = undefined;

    // 播放停用音效
    eventBus.emit(GameEvent.SFX_PLAY, { key: 'sfx_perception_deactivate' });

    // 发送停用事件
    eventBus.emit(GameEvent.PERCEPTION_END, {
      duration,
      pCost,
    });

    eventBus.emit(GameEvent.ABILITY_DEACTIVATE, {
      abilityId: 'depth_perception',
    });

    // 隐藏对象
    this._hideObjects();

    eventBus.emit(GameEvent.GESTURE_LONG_PRESS_END);
  }

  /**
   * 注册可揭示对象
   */
  registerRevealable(obj: IRevealableObject): void {
    this._revealableObjects.set(obj.id, obj);
  }

  /**
   * 取消注册
   */
  unregisterRevealable(id: string): void {
    this._revealableObjects.delete(id);
  }

  /**
   * 获取已揭示的对象
   */
  getRevealedObjects(): string[] {
    return [...this._state.revealedObjects];
  }

  /**
   * 获取状态
   */
  getState(): IPerceptionState {
    return { ...this._state };
  }

  /**
   * 是否激活中
   */
  isActive(): boolean {
    return this._state.isActive;
  }

  // ==================== 私有方法 ====================

  private _revealObjects(): void {
    for (const [id, obj] of this._revealableObjects) {
      if (this._checkRevealCondition(obj)) {
        this._state.revealedObjects.push(id);
        obj.onReveal?.();
      }
    }
  }

  private _hideObjects(): void {
    for (const id of this._state.revealedObjects) {
      const obj = this._revealableObjects.get(id);
      obj?.onHide?.();
    }
    this._state.revealedObjects = [];
  }

  private _checkRevealCondition(obj: IRevealableObject): boolean {
    if (!obj.condition) return true;
    return worldState.checkCondition(obj.condition);
  }

  private _startPAccumulation(): void {
    // 每隔一定时间检查P值累积
    this._pAccumulateTimer = window.setInterval(() => {
      const duration = Date.now() - (this._state.startTime || 0);
      const seconds = duration / 1000;
      
      // 每5秒发送一次P值变化提示
      if (seconds > 5 && seconds % this._config.pIntervalSeconds < 1) {
        eventBus.emit(GameEvent.UI_TOAST, {
          message: '观察者压力正在累积...',
          type: 'warning',
        });
      }
    }, 1000);
  }

  private _stopPAccumulation(): void {
    if (this._pAccumulateTimer) {
      clearInterval(this._pAccumulateTimer);
      this._pAccumulateTimer = null;
    }
  }

  private _calculatePCost(durationMs: number): number {
    const seconds = durationMs / 1000;
    
    for (const tier of this._config.pCostByDuration) {
      if (seconds <= tier.maxSeconds) {
        return tier.pCost;
      }
    }
    
    return this._config.pCostByDuration[this._config.pCostByDuration.length - 1].pCost;
  }

  private _createInitialState(): IPerceptionState {
    return {
      isActive: false,
      totalDuration: 0,
      revealedObjects: [],
    };
  }

  /**
   * 序列化
   */
  serialize(): { isUnlocked: boolean; totalDuration: number } {
    return {
      isUnlocked: this._isUnlocked,
      totalDuration: this._state.totalDuration,
    };
  }

  /**
   * 反序列化
   */
  deserialize(data: { isUnlocked: boolean; totalDuration: number }): void {
    this._isUnlocked = data.isUnlocked;
    this._state.totalDuration = data.totalDuration;
  }

  /**
   * 重置
   */
  reset(): void {
    this.deactivate();
    this._isUnlocked = false;
    this._state = this._createInitialState();
    this._revealableObjects.clear();
  }
}

/**
 * 可揭示对象接口
 */
interface IRevealableObject {
  id: string;
  condition?: IConditionConfig;
  onReveal?: () => void;
  onHide?: () => void;
}
```

### 3.2 DepthIntervention实现

```typescript
// src/systems/ability/DepthIntervention.ts

import { 
  IInterventionState, 
  IInterventionTarget, 
  IScar,
  IPosition,
  TZoneId 
} from '@/types';
import { eventBus, GameEvent } from '@/systems/EventBus';
import { worldState } from '@/systems/world';

/**
 * 深度介入配置
 */
interface IDepthInterventionConfig {
  /** 有效拖拽距离阈值 */
  dragThreshold: number;
  /** 介入动画时长 (ms) */
  animationDuration: number;
  /** P值消耗表 */
  pCostByScarType: Record<IScar['type'], number>;
}

const DEFAULT_CONFIG: IDepthInterventionConfig = {
  dragThreshold: 50,
  animationDuration: 800,
  pCostByScarType: {
    minor: 0.5,
    moderate: 1.0,
    severe: 2.0,
  },
};

/**
 * 深度介入能力
 */
export class DepthIntervention {
  private _config: IDepthInterventionConfig;
  private _state: IInterventionState;
  private _isUnlocked: boolean = false;
  private _targets: Map<string, IInterventionTarget> = new Map();
  private _currentZone: TZoneId | null = null;

  constructor(config: Partial<IDepthInterventionConfig> = {}) {
    this._config = { ...DEFAULT_CONFIG, ...config };
    this._state = this._createInitialState();
  }

  /**
   * 解锁能力
   */
  unlock(): void {
    this._isUnlocked = true;
    worldState.setFlag('ability_intervention_unlocked', true);
    
    eventBus.emit(GameEvent.ABILITY_UNLOCK, {
      abilityId: 'depth_intervention',
    });
  }

  /**
   * 是否已解锁
   */
  isUnlocked(): boolean {
    return this._isUnlocked;
  }

  /**
   * 设置当前Zone
   */
  setCurrentZone(zoneId: TZoneId): void {
    this._currentZone = zoneId;
  }

  /**
   * 注册可介入目标
   */
  registerTarget(target: IInterventionTarget): void {
    this._targets.set(target.id, target);
  }

  /**
   * 取消注册目标
   */
  unregisterTarget(id: string): void {
    this._targets.delete(id);
  }

  /**
   * 获取可介入目标列表
   */
  getAvailableTargets(): IInterventionTarget[] {
    return Array.from(this._targets.values())
      .filter(t => !this._state.completedInterventions.includes(t.id));
  }

  /**
   * 选中目标
   */
  selectTarget(targetId: string): boolean {
    if (!this._isUnlocked) return false;

    const target = this._targets.get(targetId);
    if (!target) return false;

    // 检查是否已介入过
    if (this._state.completedInterventions.includes(targetId)) {
      return false;
    }

    this._state.selectedTarget = target;

    eventBus.emit(GameEvent.INTERVENTION_TARGET_SELECT, {
      targetId,
      target,
    });

    return true;
  }

  /**
   * 取消选中
   */
  deselectTarget(): void {
    this._state.selectedTarget = undefined;
    this._state.dragStart = undefined;
    this._state.dragCurrent = undefined;
    this._state.isIntervening = false;
  }

  /**
   * 开始拖拽
   */
  startDrag(position: IPosition): void {
    if (!this._state.selectedTarget) return;

    this._state.isIntervening = true;
    this._state.dragStart = position;
    this._state.dragCurrent = position;

    eventBus.emit(GameEvent.GESTURE_DRAG_START, {
      position,
      targetId: this._state.selectedTarget.id,
    });

    eventBus.emit(GameEvent.INTERVENTION_START, {
      targetId: this._state.selectedTarget.id,
    });
  }

  /**
   * 更新拖拽位置
   */
  updateDrag(position: IPosition): void {
    if (!this._state.isIntervening) return;

    this._state.dragCurrent = position;

    // 计算拖拽方向
    const direction = this._calculateDragDirection();

    eventBus.emit(GameEvent.INTERVENTION_DRAG_UPDATE, {
      position,
      direction,
      isValid: this._isValidDrag(),
    });
  }

  /**
   * 结束拖拽
   */
  async endDrag(): Promise<boolean> {
    if (!this._state.isIntervening || !this._state.selectedTarget) {
      this.deselectTarget();
      return false;
    }

    eventBus.emit(GameEvent.GESTURE_DRAG_END);

    // 检查拖拽是否有效
    if (!this._isValidDrag()) {
      this.deselectTarget();
      return false;
    }

    // 执行介入
    const success = await this._executeIntervention();

    this.deselectTarget();

    return success;
  }

  /**
   * 获取状态
   */
  getState(): IInterventionState {
    return { ...this._state };
  }

  // ==================== 私有方法 ====================

  private async _executeIntervention(): Promise<boolean> {
    const target = this._state.selectedTarget!;

    // 记录已完成
    this._state.completedInterventions.push(target.id);

    // 创建伤痕
    const scar = worldState.addScar({
      zoneId: this._currentZone!,
      type: target.scarLevel,
      targetId: target.id,
      description: target.effect,
    });

    // 增加P值
    const pCost = this._config.pCostByScarType[target.scarLevel];
    worldState.addP(pCost, 'depth_intervention');

    // 标记Zone使用过深度能力
    if (this._currentZone) {
      worldState.getZoneState(this._currentZone).usedDepthAbility = true;
    }

    // 播放音效和动画
    eventBus.emit(GameEvent.SFX_PLAY, { key: 'sfx_intervention' });

    // 等待动画完成
    await this._playInterventionAnimation(target, scar);

    // 发送完成事件
    eventBus.emit(GameEvent.INTERVENTION_COMPLETE, {
      targetId: target.id,
      scar,
      pCost,
    });

    worldState.setFlag('ability_intervention_used', true);

    return true;
  }

  private async _playInterventionAnimation(
    target: IInterventionTarget, 
    scar: IScar
  ): Promise<void> {
    return new Promise(resolve => {
      eventBus.emit(GameEvent.INTERVENTION_ANIMATION_START, {
        target,
        scar,
        duration: this._config.animationDuration,
      });

      setTimeout(resolve, this._config.animationDuration);
    });
  }

  private _calculateDragDirection(): string | null {
    if (!this._state.dragStart || !this._state.dragCurrent) return null;

    const dx = this._state.dragCurrent.x - this._state.dragStart.x;
    const dy = this._state.dragCurrent.y - this._state.dragStart.y;

    const angle = Math.atan2(dy, dx) * (180 / Math.PI);

    if (angle >= -45 && angle < 45) return 'right';
    if (angle >= 45 && angle < 135) return 'down';
    if (angle >= -135 && angle < -45) return 'up';
    return 'left';
  }

  private _isValidDrag(): boolean {
    if (!this._state.dragStart || !this._state.dragCurrent || !this._state.selectedTarget) {
      return false;
    }

    // 计算拖拽距离
    const dx = this._state.dragCurrent.x - this._state.dragStart.x;
    const dy = this._state.dragCurrent.y - this._state.dragStart.y;
    const distance = Math.sqrt(dx * dx + dy * dy);

    if (distance < this._config.dragThreshold) {
      return false;
    }

    // 检查方向是否有效
    const direction = this._calculateDragDirection();
    return direction !== null && this._state.selectedTarget.directions.includes(direction as any);
  }

  private _createInitialState(): IInterventionState {
    return {
      isIntervening: false,
      completedInterventions: [],
    };
  }

  /**
   * 序列化
   */
  serialize(): { isUnlocked: boolean; completedInterventions: string[] } {
    return {
      isUnlocked: this._isUnlocked,
      completedInterventions: [...this._state.completedInterventions],
    };
  }

  /**
   * 反序列化
   */
  deserialize(data: { isUnlocked: boolean; completedInterventions: string[] }): void {
    this._isUnlocked = data.isUnlocked;
    this._state.completedInterventions = [...data.completedInterventions];
  }

  /**
   * 重置
   */
  reset(): void {
    this.deselectTarget();
    this._isUnlocked = false;
    this._state = this._createInitialState();
    this._targets.clear();
  }
}
```

## 4. 时间干预 (Time Intervention)

### 4.1 机制设计

```
时间节点生成:
┌────────────────┐
│ Zone切换       │ ─────────→ 自动创建节点
├────────────────┤
│ 关键对话完成   │ ─────────→ 自动创建节点
├────────────────┤
│ 能力使用后     │ ─────────→ 自动创建节点
├────────────────┤
│ 手动存档       │ ─────────→ 创建标记节点
└────────────────┘

回溯流程:
┌────────────────┐
│ 打开时间线UI   │
└───────┬────────┘
        │
        ▼
┌────────────────┐
│ 显示节点列表   │ ─────────→ 按时间排序，显示Zone、时间
└───────┬────────┘
        │ 选择节点
        ▼
┌────────────────┐
│ 确认回溯       │ ─────────→ 显示P值消耗、污染警告
└───────┬────────┘
        │ 确认
        ▼
┌────────────────┐
│ 执行回溯       │ ─────────→ 状态回滚（保留伤痕）
└───────┬────────┘            P值增加 + 创建污染
        │
        ▼
┌────────────────┐
│ 完成回溯       │ ─────────→ 传送到节点位置
└────────────────┘
```

### 4.2 TimeNodeManager实现

```typescript
// src/systems/ability/TimeNodeManager.ts

import { ITimeNode, TZoneId, IWorldStateData, IPosition } from '@/types';
import { eventBus, GameEvent } from '@/systems/EventBus';
import { worldState } from '@/systems/world';

/**
 * 时间节点管理器配置
 */
interface ITimeNodeManagerConfig {
  /** 最大节点数量 */
  maxNodes: number;
  /** 自动节点最小间隔 (ms) */
  minAutoInterval: number;
  /** P值计算表 */
  pCostByDistance: Array<{ maxMinutes: number; pCost: number }>;
}

const DEFAULT_CONFIG: ITimeNodeManagerConfig = {
  maxNodes: 20,
  minAutoInterval: 60000, // 1分钟
  pCostByDistance: [
    { maxMinutes: 5, pCost: 1.0 },
    { maxMinutes: 15, pCost: 2.0 },
    { maxMinutes: 30, pCost: 3.0 },
    { maxMinutes: Infinity, pCost: 5.0 },
  ],
};

/**
 * 时间节点管理器
 */
export class TimeNodeManager {
  private _config: ITimeNodeManagerConfig;
  private _nodes: ITimeNode[] = [];
  private _idCounter: number = 0;
  private _lastAutoNodeTime: number = 0;

  constructor(config: Partial<ITimeNodeManagerConfig> = {}) {
    this._config = { ...DEFAULT_CONFIG, ...config };
    this._setupEventListeners();
  }

  /**
   * 创建时间节点
   */
  createNode(params: {
    type: ITimeNode['type'];
    label: string;
    zoneId: TZoneId;
    playerPosition: IPosition;
  }): ITimeNode {
    const now = Date.now();

    // 检查自动节点间隔
    if (params.type === 'auto' && now - this._lastAutoNodeTime < this._config.minAutoInterval) {
      return this._nodes[this._nodes.length - 1]; // 返回最近的节点
    }

    const node: ITimeNode = {
      id: `node_${++this._idCounter}`,
      timestamp: now,
      zoneId: params.zoneId,
      type: params.type,
      label: params.label,
      worldSnapshot: worldState.serialize(),
      playerPosition: params.playerPosition,
      gameTime: this._calculateGameTime(),
      canRewind: true,
      rewindPCost: 0, // 后续计算
    };

    this._nodes.push(node);
    
    if (params.type === 'auto') {
      this._lastAutoNodeTime = now;
    }

    // 限制节点数量
    if (this._nodes.length > this._config.maxNodes) {
      this._removeOldestNode();
    }

    eventBus.emit(GameEvent.TIME_NODE_CREATED, {
      nodeId: node.id,
      timestamp: node.timestamp,
      zoneId: node.zoneId,
    });

    return node;
  }

  /**
   * 获取所有节点
   */
  getNodes(): ITimeNode[] {
    return [...this._nodes];
  }

  /**
   * 获取可回溯节点
   */
  getRewindableNodes(): ITimeNode[] {
    const currentTime = Date.now();
    
    return this._nodes
      .filter(n => n.canRewind && n.timestamp < currentTime)
      .map(n => ({
        ...n,
        rewindPCost: this._calculateRewindPCost(n.timestamp),
      }))
      .sort((a, b) => b.timestamp - a.timestamp); // 最新的在前
  }

  /**
   * 获取节点
   */
  getNode(nodeId: string): ITimeNode | undefined {
    return this._nodes.find(n => n.id === nodeId);
  }

  /**
   * 计算回溯P值消耗
   */
  calculatePCost(nodeId: string): number {
    const node = this.getNode(nodeId);
    if (!node) return 0;
    
    return this._calculateRewindPCost(node.timestamp);
  }

  /**
   * 标记节点为不可回溯
   */
  markUnrewindable(nodeId: string): void {
    const node = this.getNode(nodeId);
    if (node) {
      node.canRewind = false;
    }
  }

  // ==================== 私有方法 ====================

  private _setupEventListeners(): void {
    // Zone进入时创建自动节点
    eventBus.on(GameEvent.ZONE_ENTER, (data: { zoneId: TZoneId }) => {
      // 节点创建由外部控制，这里只记录
    });

    // 关键对话结束时创建节点
    eventBus.on(GameEvent.DIALOGUE_END, (data: { dialogueId: string }) => {
      // 检查是否为关键对话
    });
  }

  private _calculateRewindPCost(targetTimestamp: number): number {
    const currentTime = Date.now();
    const diffMs = currentTime - targetTimestamp;
    const diffMinutes = diffMs / (1000 * 60);

    for (const tier of this._config.pCostByDistance) {
      if (diffMinutes <= tier.maxMinutes) {
        return tier.pCost;
      }
    }

    return this._config.pCostByDistance[this._config.pCostByDistance.length - 1].pCost;
  }

  private _calculateGameTime(): number {
    // TODO: 实际游戏时间计算
    return Date.now();
  }

  private _removeOldestNode(): void {
    // 移除最旧的自动节点
    const autoNodes = this._nodes.filter(n => n.type === 'auto');
    if (autoNodes.length > 0) {
      const oldest = autoNodes[0];
      this._nodes = this._nodes.filter(n => n.id !== oldest.id);
    }
  }

  /**
   * 序列化
   */
  serialize(): ITimeNode[] {
    return [...this._nodes];
  }

  /**
   * 反序列化
   */
  deserialize(data: ITimeNode[]): void {
    this._nodes = [...data];
    const maxId = Math.max(0, ...data.map(n => parseInt(n.id.split('_')[1]) || 0));
    this._idCounter = maxId;
  }

  /**
   * 重置
   */
  reset(): void {
    this._nodes = [];
    this._idCounter = 0;
    this._lastAutoNodeTime = 0;
  }
}
```

```typescript
// src/systems/ability/TimeIntervention.ts

import { ITimeInterventionState, ITimeNode, IContamination, TZoneId } from '@/types';
import { eventBus, GameEvent } from '@/systems/EventBus';
import { worldState } from '@/systems/world';
import { TimeNodeManager } from './TimeNodeManager';

/**
 * 时间干预能力
 */
export class TimeIntervention {
  private _state: ITimeInterventionState;
  private _nodeManager: TimeNodeManager;
  private _isUnlocked: boolean = false;
  private _rewindInProgress: boolean = false;

  constructor(nodeManager: TimeNodeManager) {
    this._nodeManager = nodeManager;
    this._state = this._createInitialState();
  }

  /**
   * 解锁能力
   */
  unlock(): void {
    this._isUnlocked = true;
    worldState.setFlag('ability_time_unlocked', true);
    
    eventBus.emit(GameEvent.ABILITY_UNLOCK, {
      abilityId: 'time_intervention',
    });
  }

  /**
   * 是否已解锁
   */
  isUnlocked(): boolean {
    return this._isUnlocked;
  }

  /**
   * 显示时间线UI
   */
  showTimeline(): void {
    if (!this._isUnlocked) return;

    this._state.isUIVisible = true;
    this._state.availableNodes = this._nodeManager.getRewindableNodes();

    eventBus.emit(GameEvent.UI_PANEL_OPEN, {
      panelType: 'time_line',
    });
  }

  /**
   * 隐藏时间线UI
   */
  hideTimeline(): void {
    this._state.isUIVisible = false;
    this._state.selectedNode = undefined;

    eventBus.emit(GameEvent.UI_PANEL_CLOSE, {
      panelType: 'time_line',
    });
  }

  /**
   * 选择节点
   */
  selectNode(nodeId: string): void {
    const node = this._nodeManager.getNode(nodeId);
    if (node) {
      this._state.selectedNode = node;
    }
  }

  /**
   * 取消选择
   */
  deselectNode(): void {
    this._state.selectedNode = undefined;
  }

  /**
   * 获取选中节点的回溯信息
   */
  getRewindInfo(): { pCost: number; warning: string } | null {
    if (!this._state.selectedNode) return null;

    const pCost = this._nodeManager.calculatePCost(this._state.selectedNode.id);
    let warning = '';

    if (pCost >= 3) {
      warning = '⚠️ 大范围回溯将产生严重污染';
    } else if (pCost >= 2) {
      warning = '⚠️ 回溯将产生中等污染';
    } else {
      warning = '回溯将产生轻微污染';
    }

    return { pCost, warning };
  }

  /**
   * 执行回溯
   */
  async rewind(): Promise<boolean> {
    if (!this._state.selectedNode || this._rewindInProgress) {
      return false;
    }

    const targetNode = this._state.selectedNode;
    const currentNodeId = this._nodeManager.getNodes().slice(-1)[0]?.id;

    this._rewindInProgress = true;

    // 计算P值消耗
    const pCost = this._nodeManager.calculatePCost(targetNode.id);

    // 发送开始事件
    eventBus.emit(GameEvent.TIME_REWIND_START, {
      fromNodeId: currentNodeId,
      toNodeId: targetNode.id,
      pCost,
    });

    // 播放回溯动画和音效
    eventBus.emit(GameEvent.SFX_PLAY, { key: 'sfx_time_rewind' });
    await this._playRewindAnimation();

    // 保存当前伤痕（回溯后保留）
    const currentScars = worldState.getAllScars();

    // 恢复世界状态
    worldState.deserialize(targetNode.worldSnapshot);

    // 重新添加伤痕（伤痕不随回溯消失）
    for (const scar of currentScars) {
      if (!worldState.getAllScars().find(s => s.id === scar.id)) {
        worldState.addScar(scar);
      }
    }

    // 增加P值
    worldState.addP(pCost, 'time_intervention');

    // 创建污染
    this._createContamination(currentNodeId || '', targetNode.id, pCost);

    // 记录回溯历史
    this._state.rewindHistory.push({
      fromNodeId: currentNodeId || '',
      toNodeId: targetNode.id,
      timestamp: Date.now(),
    });

    // 发送完成事件
    eventBus.emit(GameEvent.TIME_REWIND_COMPLETE, {
      targetNode,
      pCost,
    });

    worldState.setFlag('ability_time_used', true);

    // 隐藏UI
    this.hideTimeline();

    this._rewindInProgress = false;

    return true;
  }

  /**
   * 获取状态
   */
  getState(): ITimeInterventionState {
    return { ...this._state };
  }

  /**
   * 获取节点管理器
   */
  getNodeManager(): TimeNodeManager {
    return this._nodeManager;
  }

  // ==================== 私有方法 ====================

  private async _playRewindAnimation(): Promise<void> {
    return new Promise(resolve => {
      eventBus.emit(GameEvent.TIME_REWIND_ANIMATION, {
        duration: 1500,
      });
      setTimeout(resolve, 1500);
    });
  }

  private _createContamination(fromNodeId: string, toNodeId: string, pCost: number): void {
    // 根据P值确定污染强度
    let intensity: number;
    let effects: string[];

    if (pCost >= 5) {
      intensity = 0.8;
      effects = ['memory_drift', 'npc_glitch', 'area_distortion'];
    } else if (pCost >= 3) {
      intensity = 0.5;
      effects = ['env_patches', 'memory_repeat'];
    } else if (pCost >= 2) {
      intensity = 0.3;
      effects = ['npc_repeat'];
    } else {
      intensity = 0.1;
      effects = ['minor_glitch'];
    }

    // 确定受影响Zone
    const targetNode = this._nodeManager.getNode(toNodeId);
    const affectedZones: TZoneId[] = [];
    
    if (targetNode) {
      affectedZones.push(targetNode.zoneId);
      // 可能影响相邻Zone
    }

    worldState.addContamination({
      sourceNodeId: fromNodeId,
      targetNodeId: toNodeId,
      intensity,
      affectedZones,
      effects,
    });
  }

  private _createInitialState(): ITimeInterventionState {
    return {
      isUIVisible: false,
      availableNodes: [],
      isRewinding: false,
      rewindHistory: [],
    };
  }

  /**
   * 序列化
   */
  serialize(): { isUnlocked: boolean; rewindHistory: ITimeInterventionState['rewindHistory'] } {
    return {
      isUnlocked: this._isUnlocked,
      rewindHistory: [...this._state.rewindHistory],
    };
  }

  /**
   * 反序列化
   */
  deserialize(data: { isUnlocked: boolean; rewindHistory: ITimeInterventionState['rewindHistory'] }): void {
    this._isUnlocked = data.isUnlocked;
    this._state.rewindHistory = [...data.rewindHistory];
  }

  /**
   * 重置
   */
  reset(): void {
    this.hideTimeline();
    this._isUnlocked = false;
    this._state = this._createInitialState();
    this._rewindInProgress = false;
  }
}
```

```typescript
// src/systems/ability/AbilitySystem.ts

import { AbilityType, IPosition, TZoneId } from '@/types';
import { DepthPerception } from './DepthPerception';
import { DepthIntervention } from './DepthIntervention';
import { TimeIntervention } from './TimeIntervention';
import { TimeNodeManager } from './TimeNodeManager';
import { eventBus, GameEvent } from '@/systems/EventBus';
import { worldState } from '@/systems/world';

/**
 * 能力系统 - 单例
 */
export class AbilitySystem {
  private static _instance: AbilitySystem;

  private _perception: DepthPerception;
  private _intervention: DepthIntervention;
  private _timeIntervention: TimeIntervention;
  private _nodeManager: TimeNodeManager;
  private _currentZone: TZoneId | null = null;
  private _playerPosition: IPosition = { x: 0, y: 0 };
  private _initialized: boolean = false;

  private constructor() {
    this._perception = new DepthPerception();
    this._intervention = new DepthIntervention();
    this._nodeManager = new TimeNodeManager();
    this._timeIntervention = new TimeIntervention(this._nodeManager);
  }

  /**
   * 获取单例
   */
  static getInstance(): AbilitySystem {
    if (!AbilitySystem._instance) {
      AbilitySystem._instance = new AbilitySystem();
    }
    return AbilitySystem._instance;
  }

  /**
   * 初始化
   */
  initialize(): void {
    if (this._initialized) return;

    this._setupEventListeners();
    this._initialized = true;
  }

  /**
   * 更新当前位置信息
   */
  updateContext(zoneId: TZoneId, playerPosition: IPosition): void {
    this._currentZone = zoneId;
    this._playerPosition = playerPosition;
    this._intervention.setCurrentZone(zoneId);
  }

  // ==================== 能力解锁 ====================

  /**
   * 解锁能力
   */
  unlockAbility(type: AbilityType): void {
    switch (type) {
      case AbilityType.DEPTH_PERCEPTION:
        this._perception.unlock();
        break;
      case AbilityType.DEPTH_INTERVENTION:
        this._intervention.unlock();
        break;
      case AbilityType.TIME_INTERVENTION:
        this._timeIntervention.unlock();
        break;
    }
  }

  /**
   * 检查能力是否解锁
   */
  isAbilityUnlocked(type: AbilityType): boolean {
    switch (type) {
      case AbilityType.DEPTH_PERCEPTION:
        return this._perception.isUnlocked();
      case AbilityType.DEPTH_INTERVENTION:
        return this._intervention.isUnlocked();
      case AbilityType.TIME_INTERVENTION:
        return this._timeIntervention.isUnlocked();
      default:
        return false;
    }
  }

  /**
   * 获取所有已解锁能力
   */
  getUnlockedAbilities(): AbilityType[] {
    const abilities: AbilityType[] = [];
    
    if (this._perception.isUnlocked()) abilities.push(AbilityType.DEPTH_PERCEPTION);
    if (this._intervention.isUnlocked()) abilities.push(AbilityType.DEPTH_INTERVENTION);
    if (this._timeIntervention.isUnlocked()) abilities.push(AbilityType.TIME_INTERVENTION);
    
    return abilities;
  }

  // ==================== 深度感知接口 ====================

  get perception(): DepthPerception {
    return this._perception;
  }

  /**
   * 开始深度感知蓄力
   */
  startPerceptionCharge(): void {
    this._perception.startCharge();
  }

  /**
   * 停止深度感知
   */
  stopPerception(): void {
    this._perception.deactivate();
  }

  /**
   * 深度感知是否激活
   */
  isPerceptionActive(): boolean {
    return this._perception.isActive();
  }

  // ==================== 深度介入接口 ====================

  get intervention(): DepthIntervention {
    return this._intervention;
  }

  /**
   * 选择介入目标
   */
  selectInterventionTarget(targetId: string): boolean {
    return this._intervention.selectTarget(targetId);
  }

  /**
   * 开始介入拖拽
   */
  startInterventionDrag(position: IPosition): void {
    this._intervention.startDrag(position);
  }

  /**
   * 更新介入拖拽
   */
  updateInterventionDrag(position: IPosition): void {
    this._intervention.updateDrag(position);
  }

  /**
   * 结束介入拖拽
   */
  async endInterventionDrag(): Promise<boolean> {
    return this._intervention.endDrag();
  }

  // ==================== 时间干预接口 ====================

  get timeIntervention(): TimeIntervention {
    return this._timeIntervention;
  }

  /**
   * 显示时间线
   */
  showTimeline(): void {
    this._timeIntervention.showTimeline();
  }

  /**
   * 隐藏时间线
   */
  hideTimeline(): void {
    this._timeIntervention.hideTimeline();
  }

  /**
   * 执行时间回溯
   */
  async rewindTime(): Promise<boolean> {
    return this._timeIntervention.rewind();
  }

  /**
   * 创建时间节点
   */
  createTimeNode(type: 'auto' | 'manual' | 'event', label: string): void {
    if (!this._currentZone) return;

    this._nodeManager.createNode({
      type,
      label,
      zoneId: this._currentZone,
      playerPosition: this._playerPosition,
    });
  }

  // ==================== 序列化 ====================

  /**
   * 序列化
   */
  serialize() {
    return {
      perception: this._perception.serialize(),
      intervention: this._intervention.serialize(),
      timeIntervention: this._timeIntervention.serialize(),
      timeNodes: this._nodeManager.serialize(),
    };
  }

  /**
   * 反序列化
   */
  deserialize(data: ReturnType<typeof this.serialize>): void {
    this._perception.deserialize(data.perception);
    this._intervention.deserialize(data.intervention);
    this._timeIntervention.deserialize(data.timeIntervention);
    this._nodeManager.deserialize(data.timeNodes);
  }

  /**
   * 重置
   */
  reset(): void {
    this._perception.reset();
    this._intervention.reset();
    this._timeIntervention.reset();
    this._nodeManager.reset();
    this._currentZone = null;
    this._playerPosition = { x: 0, y: 0 };
  }

  // ==================== 私有方法 ====================

  private _setupEventListeners(): void {
    // Zone进入时自动创建节点
    eventBus.on(GameEvent.ZONE_ENTER, (data: { zoneId: TZoneId }) => {
      if (this._timeIntervention.isUnlocked()) {
        this.createTimeNode('auto', `进入 ${data.zoneId}`);
      }
    });

    // 关键对话后创建节点
    eventBus.on(GameEvent.DIALOGUE_END, (data: { dialogueId: string }) => {
      // 检查是否为关键对话
      if (this._isKeyDialogue(data.dialogueId) && this._timeIntervention.isUnlocked()) {
        this.createTimeNode('event', `对话 ${data.dialogueId}`);
      }
    });

    // 能力使用后创建节点
    eventBus.on(GameEvent.INTERVENTION_COMPLETE, () => {
      if (this._timeIntervention.isUnlocked()) {
        this.createTimeNode('event', '深度介入');
      }
    });
  }

  private _isKeyDialogue(dialogueId: string): boolean {
    // TODO: 根据配置判断是否为关键对话
    return dialogueId.includes('KEY') || dialogueId.includes('FIRST');
  }
}
```
