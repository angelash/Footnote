# 《备注 / Footnote》场景与资源管理设计

> **文档版本**：v1.0  
> **文档性质**：子系统设计规范  
> **适用对象**：游戏开发  
> **最后更新**：2025-12-24

## 1. 系统概述

### 1.1 职责划分

| 系统 | 职责 |
|------|------|
| **SceneAssembler** | 解析YAML配置，创建Phaser GameObjects |
| **ZoneManager** | Zone切换、资源协调、状态管理 |
| **ResourceManager** | 资源加载、缓存、卸载策略 |
| **TransitionManager** | 场景过渡动画 |

### 1.2 系统架构


## 2. SceneAssembler详细设计

### 2.1 完整实现

```typescript
// src/systems/scene/SceneAssembler.ts

import Phaser from 'phaser';
import {
  ISceneConfig,
  ISceneObjectConfig,
  ISceneBackgroundConfig,
  IExitConfig,
  ISceneAssemblerCallbacks,
  IAssembledScene,
  IConditionConfig,
} from '@/types';
import { worldState } from '@/systems/world';
import { eventBus, GameEvent } from '@/systems/EventBus';

/**
 * 场景组装器
 */
export class SceneAssembler {
  private _scene: Phaser.Scene;
  private _config: ISceneConfig | null = null;
  private _callbacks: ISceneAssemblerCallbacks;
  private _objects: Map<string, Phaser.GameObjects.GameObject[]> = new Map();
  private _labels: Map<string, Phaser.GameObjects.Text> = new Map();
  private _exits: Phaser.GameObjects.Zone[] = [];
  private _background: Phaser.GameObjects.Image | null = null;
  private _perceptionContainer: Phaser.GameObjects.Container | null = null;

  constructor(scene: Phaser.Scene, callbacks: ISceneAssemblerCallbacks = {}) {
    this._scene = scene;
    this._callbacks = callbacks;
  }

  /**
   * 构建场景
   */
  build(config: ISceneConfig): IAssembledScene {
    this._config = config;
    
    const assembled: IAssembledScene = {
      background: null,
      objects: [],
      exits: [],
    };

    // 1. 创建背景
    if (config.background) {
      assembled.background = this._createBackground(config.background);
      this._background = assembled.background;
    }

    // 2. 创建主要对象
    for (const objConfig of config.objects) {
      // 检查条件
      if (objConfig.condition && !this._checkCondition(objConfig.condition)) {
        continue;
      }
      
      // 跳过深度感知专属对象
      if (objConfig.perceptionOnly) {
        continue;
      }

      const objects = this._createObject(objConfig);
      assembled.objects.push(...objects);
      this._objects.set(objConfig.id, objects);
    }

    // 3. 创建出口
    if (config.exits) {
      for (const exitConfig of config.exits) {
        if (exitConfig.condition && !this._checkCondition(exitConfig.condition)) {
          continue;
        }
        const exit = this._createExit(exitConfig);
        assembled.exits.push(exit);
        this._exits.push(exit);
      }
    }

    // 4. 创建深度感知层
    if (config.perceptionLayer) {
      assembled.perceptionLayer = this._createPerceptionLayer(config.perceptionLayer);
      this._perceptionContainer = assembled.perceptionLayer;
    }

    // 5. 设置音频
    if (config.audio) {
      this._setupAudio(config.audio);
    }

    return assembled;
  }

  /**
   * 销毁场景
   */
  destroy(): void {
    // 销毁所有对象
    for (const objects of this._objects.values()) {
      objects.forEach(obj => obj.destroy());
    }
    this._objects.clear();

    // 销毁标签
    for (const label of this._labels.values()) {
      label.destroy();
    }
    this._labels.clear();

    // 销毁出口
    this._exits.forEach(exit => exit.destroy());
    this._exits = [];

    // 销毁背景
    this._background?.destroy();
    this._background = null;

    // 销毁感知层
    this._perceptionContainer?.destroy();
    this._perceptionContainer = null;

    this._config = null;
  }

  /**
   * 获取对象
   */
  getObject(id: string): Phaser.GameObjects.GameObject | undefined {
    return this._objects.get(id)?.[0];
  }

  /**
   * 更新对象属性
   */
  updateObject(id: string, props: Partial<ISceneObjectConfig>): void {
    const objects = this._objects.get(id);
    if (!objects || objects.length === 0) return;

    const mainObj = objects[0] as Phaser.GameObjects.Image | Phaser.GameObjects.Sprite;

    if (props.x !== undefined) mainObj.setX(props.x);
    if (props.y !== undefined) mainObj.setY(props.y);
    if (props.scale !== undefined) mainObj.setScale(props.scale);
    if (props.alpha !== undefined) mainObj.setAlpha(props.alpha);
    if (props.visible !== undefined) mainObj.setVisible(props.visible);
    if (props.depth !== undefined) mainObj.setDepth(props.depth);
  }

  /**
   * 显示/隐藏深度感知层
   */
  setPerceptionLayerVisible(visible: boolean): void {
    if (this._perceptionContainer) {
      this._perceptionContainer.setVisible(visible);
      this._perceptionContainer.setAlpha(visible ? 1 : 0);
    }
  }

  /**
   * 重新检查条件并更新可见性
   */
  refreshConditions(): void {
    if (!this._config) return;

    for (const objConfig of this._config.objects) {
      if (!objConfig.condition) continue;

      const objects = this._objects.get(objConfig.id);
      if (!objects) continue;

      const shouldShow = this._checkCondition(objConfig.condition);
      objects.forEach(obj => {
        (obj as Phaser.GameObjects.Components.Visible).setVisible(shouldShow);
      });
    }
  }

  // ==================== 私有方法 ====================

  private _createBackground(config: ISceneBackgroundConfig): Phaser.GameObjects.Image {
    const bg = this._scene.add.image(config.x, config.y, config.texture);

    if (config.origin) {
      bg.setOrigin(config.origin[0], config.origin[1]);
    }

    if (config.displaySize) {
      bg.setDisplaySize(config.displaySize[0], config.displaySize[1]);
    }

    if (config.scrollFactor) {
      bg.setScrollFactor(config.scrollFactor[0], config.scrollFactor[1]);
    }

    bg.setDepth(-1000);

    return bg;
  }

  private _createObject(config: ISceneObjectConfig): Phaser.GameObjects.GameObject[] {
    const created: Phaser.GameObjects.GameObject[] = [];

    // 创建主对象
    let display: Phaser.GameObjects.Image | Phaser.GameObjects.Sprite;

    if (config.type === 'sprite' && config.animation) {
      display = this._scene.add.sprite(config.x, config.y, config.texture);
      this._setupAnimation(display as Phaser.GameObjects.Sprite, config.animation);
    } else {
      display = this._scene.add.image(config.x, config.y, config.texture);
    }

    // 设置基础属性
    if (config.origin) {
      display.setOrigin(config.origin[0], config.origin[1]);
    }
    if (typeof config.scale === 'number') {
      display.setScale(config.scale);
    }
    if (typeof config.alpha === 'number') {
      display.setAlpha(config.alpha);
    }
    if (typeof config.rotation === 'number') {
      display.setRotation(config.rotation);
    }
    if (config.tint !== undefined) {
      display.setTint(config.tint as number);
    }
    if (config.visible === false) {
      display.setVisible(false);
    }

    // 设置深度
    if (typeof config.depth === 'number') {
      display.setDepth(config.depth);
    } else {
      display.setDepth(config.y); // Y-sort
    }

    // 设置数据
    display.setData('objectId', config.id);
    display.setData('config', config);

    created.push(display);

    // 设置交互
    if (config.interactive) {
      this._setupInteractive(display, config);
    }

    // 创建标签
    if (config.label) {
      const label = this._createLabel(config);
      created.push(label);
      this._labels.set(config.id, label);
    }

    return created;
  }

  private _setupAnimation(
    sprite: Phaser.GameObjects.Sprite,
    config: ISceneObjectConfig['animation']
  ): void {
    if (!config) return;

    const animKey = config.key;

    // 检查动画是否已存在
    if (!this._scene.anims.exists(animKey)) {
      const texture = sprite.texture.key;
      const frameCount = sprite.texture.frameTotal - 1;

      this._scene.anims.create({
        key: animKey,
        frames: this._scene.anims.generateFrameNumbers(texture, {
          start: config.frames?.start ?? 0,
          end: config.frames?.end ?? frameCount,
        }),
        frameRate: config.frameRate ?? 10,
        repeat: config.repeat ?? -1,
      });
    }

    // 播放动画
    if (config.delay) {
      this._scene.time.delayedCall(config.delay, () => {
        sprite.play(animKey);
      });
    } else {
      sprite.play(animKey);
    }
  }

  private _setupInteractive(
    display: Phaser.GameObjects.Image | Phaser.GameObjects.Sprite,
    config: ISceneObjectConfig
  ): void {
    const interactive = config.interactive!;

    display.setInteractive({
      useHandCursor: interactive.cursor !== false,
    });

    // 设置测试ID
    if (interactive.testid) {
      display.setData('testid', interactive.testid);
    }

    // 高亮效果
    display.on('pointerover', () => {
      if (interactive.highlight) {
        display.setTint(interactive.highlight.color as number || 0xffffaa);
        if (interactive.highlight.scale) {
          display.setScale(config.scale! * interactive.highlight.scale);
        }
      } else {
        display.setTint(0xffffaa);
      }

      // 显示标签
      const label = this._labels.get(config.id);
      if (label) {
        label.setVisible(true);
      }
    });

    display.on('pointerout', () => {
      display.clearTint();
      if (typeof config.scale === 'number') {
        display.setScale(config.scale);
      }

      // 隐藏标签
      const label = this._labels.get(config.id);
      if (label) {
        label.setVisible(false);
      }
    });

    // 点击处理
    display.on('pointerdown', () => {
      this._handleInteraction(config);
    });
  }

  private _handleInteraction(config: ISceneObjectConfig): void {
    const action = config.interactive?.action;
    if (!action) return;

    eventBus.emit(GameEvent.INTERACT_START, {
      objectId: config.id,
      actionType: action.type,
    });

    switch (action.type) {
      case 'dialogue':
        this._callbacks.onDialogue?.(
          action.dialogueId || '',
          action.speaker,
          action.text
        );
        break;

      case 'card':
        this._callbacks.onCard?.(action.cardId || '');
        break;

      case 'gotoZone':
        this._callbacks.onGotoZone?.(action.zoneId!);
        break;

      case 'trigger':
        this._callbacks.onTrigger?.(action.triggerId || '', action.params);
        break;

      case 'custom':
        this._callbacks.onCustom?.(action.callback || '', action.params);
        break;
    }

    eventBus.emit(GameEvent.INTERACT_END, {
      objectId: config.id,
    });
  }

  private _createLabel(config: ISceneObjectConfig): Phaser.GameObjects.Text {
    const offsetX = config.labelOffset?.[0] ?? 0;
    const offsetY = config.labelOffset?.[1] ?? 60;

    const label = this._scene.add.text(
      config.x + offsetX,
      config.y + offsetY,
      config.label!,
      {
        fontSize: '24px',
        fontFamily: 'sans-serif',
        color: '#ffffff',
        backgroundColor: 'rgba(0, 0, 0, 0.7)',
        padding: { x: 8, y: 4 },
      }
    );

    label.setOrigin(0.5, 0);
    label.setDepth(10000);
    label.setVisible(false);

    return label;
  }

  private _createExit(config: IExitConfig): Phaser.GameObjects.Zone {
    const zone = this._scene.add.zone(
      config.position[0],
      config.position[1],
      100,
      100
    );

    zone.setData('exitId', config.id);
    zone.setData('target', config.target);
    zone.setData('targetSpawn', config.targetSpawn);

    // 设置物理碰撞区域（如果使用Arcade Physics）
    this._scene.physics.add.existing(zone, true);

    return zone;
  }

  private _createPerceptionLayer(
    config: ISceneConfig['perceptionLayer']
  ): Phaser.GameObjects.Container {
    if (!config) return this._scene.add.container(0, 0);

    const container = this._scene.add.container(0, 0);
    container.setVisible(false);
    container.setDepth(5000);

    // 创建深度感知专属对象
    for (const objConfig of config.objects) {
      const objects = this._createObject(objConfig);
      objects.forEach(obj => container.add(obj));
    }

    // 应用视觉效果
    if (config.effects) {
      // 效果在GameScene中通过shader处理
      container.setData('effects', config.effects);
    }

    return container;
  }

  private _setupAudio(config: ISceneConfig['audio']): void {
    if (!config) return;

    if (config.bgm) {
      eventBus.emit(GameEvent.BGM_PLAY, {
        key: config.bgm,
        crossfade: config.crossfade,
        fadeDuration: config.fadeDuration,
      });
    }

    if (config.ambience) {
      eventBus.emit(GameEvent.AMBIENCE_PLAY, {
        key: config.ambience,
      });
    }
  }

  private _checkCondition(condition: IConditionConfig): boolean {
    return worldState.checkCondition(condition);
  }
}
```

```typescript
// src/systems/scene/ZoneManager.ts

import Phaser from 'phaser';
import { 
  ISceneConfig, 
  TZoneId, 
  IPosition,
  ISceneAssemblerCallbacks 
} from '@/types';
import { SceneAssembler } from './SceneAssembler';
import { ResourceManager } from './ResourceManager';
import { TransitionManager, TransitionType } from './TransitionManager';
import { eventBus, GameEvent } from '@/systems/EventBus';
import { worldState } from '@/systems/world';

/**
 * Zone管理器配置
 */
interface IZoneManagerConfig {
  defaultTransition: TransitionType;
  transitionDuration: number;
  preloadAdjacentZones: boolean;
}

const DEFAULT_CONFIG: IZoneManagerConfig = {
  defaultTransition: TransitionType.FADE,
  transitionDuration: 500,
  preloadAdjacentZones: true,
};

/**
 * Zone管理器
 */
export class ZoneManager {
  private _scene: Phaser.Scene;
  private _config: IZoneManagerConfig;
  private _currentZoneId: TZoneId | null = null;
  private _currentConfig: ISceneConfig | null = null;
  private _assembler: SceneAssembler | null = null;
  private _resourceManager: ResourceManager;
  private _transitionManager: TransitionManager;
  private _zoneConfigs: Map<TZoneId, ISceneConfig> = new Map();
  private _callbacks: ISceneAssemblerCallbacks = {};
  private _isSwitching: boolean = false;

  constructor(
    scene: Phaser.Scene,
    config: Partial<IZoneManagerConfig> = {}
  ) {
    this._scene = scene;
    this._config = { ...DEFAULT_CONFIG, ...config };
    this._resourceManager = new ResourceManager(scene);
    this._transitionManager = new TransitionManager(scene);
  }

  /**
   * 注册Zone配置
   */
  registerZones(configs: Record<TZoneId, ISceneConfig>): void {
    for (const [id, config] of Object.entries(configs)) {
      this._zoneConfigs.set(id as TZoneId, config);
    }
  }

  /**
   * 设置交互回调
   */
  setCallbacks(callbacks: ISceneAssemblerCallbacks): void {
    this._callbacks = callbacks;
  }

  /**
   * 获取当前Zone ID
   */
  getCurrentZoneId(): TZoneId | null {
    return this._currentZoneId;
  }

  /**
   * 获取当前Zone配置
   */
  getCurrentConfig(): ISceneConfig | null {
    return this._currentConfig;
  }

  /**
   * 切换Zone
   */
  async switchZone(
    targetZoneId: TZoneId,
    options: {
      transition?: TransitionType;
      spawnPosition?: IPosition;
    } = {}
  ): Promise<boolean> {
    // 防止重复切换
    if (this._isSwitching) {
      console.warn('Zone switch already in progress');
      return false;
    }

    // 获取目标配置
    const targetConfig = this._zoneConfigs.get(targetZoneId);
    if (!targetConfig) {
      console.error(`Zone config not found: ${targetZoneId}`);
      return false;
    }

    // 检查是否解锁
    if (!worldState.isZoneUnlocked(targetZoneId)) {
      console.warn(`Zone ${targetZoneId} is locked`);
      eventBus.emit(GameEvent.UI_TOAST, {
        message: '该区域尚未解锁',
        type: 'warning',
      });
      return false;
    }

    this._isSwitching = true;

    try {
      // 发送离开事件
      if (this._currentZoneId) {
        eventBus.emit(GameEvent.ZONE_EXIT, {
          zoneId: this._currentZoneId,
          nextZoneId: targetZoneId,
        });
      }

      // 执行过渡动画（淡出）
      const transition = options.transition ?? this._config.defaultTransition;
      await this._transitionManager.out(transition, this._config.transitionDuration);

      // 销毁当前场景
      this._destroyCurrentZone();

      // 加载新Zone资源
      await this._resourceManager.preloadForZone(targetZoneId);

      // 组装新场景
      this._assembler = new SceneAssembler(this._scene, this._callbacks);
      this._assembler.build(targetConfig);

      // 更新状态
      this._currentZoneId = targetZoneId;
      this._currentConfig = targetConfig;

      // 更新WorldState
      worldState.visitZone(targetZoneId);

      // 执行过渡动画（淡入）
      await this._transitionManager.in(transition, this._config.transitionDuration);

      // 发送进入事件
      eventBus.emit(GameEvent.ZONE_ENTER, {
        zoneId: targetZoneId,
        isFirstVisit: !worldState.isZoneVisited(targetZoneId),
        isRevisit: worldState.getZoneState(targetZoneId).visitCount > 1,
      });

      // 触发进入事件
      await this._triggerZoneEvents('on_enter');

      // 预加载相邻Zone
      if (this._config.preloadAdjacentZones) {
        this._preloadAdjacentZones(targetZoneId);
      }

      return true;

    } catch (error) {
      console.error('Zone switch failed:', error);
      return false;

    } finally {
      this._isSwitching = false;
    }
  }

  /**
   * 重新加载当前Zone
   */
  async reloadCurrentZone(): Promise<boolean> {
    if (!this._currentZoneId) return false;
    return this.switchZone(this._currentZoneId);
  }

  /**
   * 刷新条件显示
   */
  refreshConditions(): void {
    this._assembler?.refreshConditions();
  }

  /**
   * 显示/隐藏深度感知层
   */
  setPerceptionMode(enabled: boolean): void {
    this._assembler?.setPerceptionLayerVisible(enabled);
  }

  /**
   * 获取场景组装器
   */
  getAssembler(): SceneAssembler | null {
    return this._assembler;
  }

  /**
   * 检查玩家是否在出口区域
   */
  checkExitCollision(playerPosition: IPosition): TZoneId | null {
    // TODO: 检查玩家位置是否与出口碰撞
    return null;
  }

  // ==================== 私有方法 ====================

  private _destroyCurrentZone(): void {
    if (this._assembler) {
      this._assembler.destroy();
      this._assembler = null;
    }

    // 卸载不需要的资源
    if (this._currentZoneId) {
      this._resourceManager.unloadUnused(
        this._currentZoneId,
        this._getAdjacentZones(this._currentZoneId)
      );
    }
  }

  private async _triggerZoneEvents(eventType: 'on_enter' | 'on_exit' | 'on_first_enter' | 'on_revisit'): Promise<void> {
    if (!this._currentConfig?.events) return;

    const events = this._currentConfig.events[eventType];
    if (!events) return;

    for (const event of events) {
      switch (event.type) {
        case 'dialogue':
          if (event.dialogueId) {
            this._callbacks.onDialogue?.(event.dialogueId, event.speaker, event.text);
          } else if (event.speaker && event.text) {
            this._callbacks.onDialogue?.('', event.speaker, event.text);
          }
          break;

        case 'card':
          if (event.cardId) {
            this._callbacks.onCard?.(event.cardId);
          }
          break;

        case 'trigger':
          if (event.triggerId) {
            this._callbacks.onTrigger?.(event.triggerId, event.params);
          }
          break;
      }
    }
  }

  private _preloadAdjacentZones(currentZoneId: TZoneId): void {
    const adjacentZones = this._getAdjacentZones(currentZoneId);
    
    // 后台预加载
    requestIdleCallback(() => {
      adjacentZones.forEach(zoneId => {
        this._resourceManager.preloadForZone(zoneId).catch(() => {
          // 预加载失败不影响游戏
        });
      });
    });
  }

  private _getAdjacentZones(zoneId: TZoneId): TZoneId[] {
    const config = this._zoneConfigs.get(zoneId);
    if (!config?.exits) return [];

    return config.exits.map(e => e.target);
  }

  /**
   * 序列化
   */
  serialize(): { currentZoneId: TZoneId | null } {
    return {
      currentZoneId: this._currentZoneId,
    };
  }

  /**
   * 反序列化
   */
  async deserialize(data: { currentZoneId: TZoneId | null }): Promise<void> {
    if (data.currentZoneId) {
      await this.switchZone(data.currentZoneId);
    }
  }
}
```

### 4.1 完整实现

```typescript
// src/systems/scene/ResourceManager.ts

import Phaser from 'phaser';
import { TZoneId } from '@/types';

/**
 * 资源类型
 */
enum ResourceType {
  IMAGE = 'image',
  SPRITESHEET = 'spritesheet',
  AUDIO = 'audio',
  YAML = 'yaml',
}

/**
 * 资源配置
 */
interface IResourceEntry {
  key: string;
  type: ResourceType;
  path: string;
  frameConfig?: Phaser.Types.Loader.FileTypes.ImageFrameConfig;
}

/**
 * 资源管理器
 */
export class ResourceManager {
  private _scene: Phaser.Scene;
  private _loadedResources: Set<string> = new Set();
  private _loadingPromises: Map<string, Promise<void>> = new Map();
  private _zoneResources: Map<TZoneId, IResourceEntry[]> = new Map();
  private _criticalResources: IResourceEntry[] = [];

  constructor(scene: Phaser.Scene) {
    this._scene = scene;
  }

  /**
   * 注册Zone资源
   */
  registerZoneResources(zoneId: TZoneId, resources: IResourceEntry[]): void {
    this._zoneResources.set(zoneId, resources);
  }

  /**
   * 注册关键资源（始终保留）
   */
  registerCriticalResources(resources: IResourceEntry[]): void {
    this._criticalResources = resources;
  }

  /**
   * 预加载Zone资源
   */
  async preloadForZone(zoneId: TZoneId): Promise<void> {
    const resources = this._zoneResources.get(zoneId);
    if (!resources) return;

    // 筛选需要加载的资源
    const toLoad = resources.filter(r => !this._loadedResources.has(r.key));
    if (toLoad.length === 0) return;

    // 检查是否正在加载
    const existingPromise = this._loadingPromises.get(zoneId);
    if (existingPromise) {
      return existingPromise;
    }

    // 创建加载Promise
    const loadPromise = this._loadResources(toLoad);
    this._loadingPromises.set(zoneId, loadPromise);

    try {
      await loadPromise;
      toLoad.forEach(r => this._loadedResources.add(r.key));
    } finally {
      this._loadingPromises.delete(zoneId);
    }
  }

  /**
   * 卸载不需要的资源
   */
  unloadUnused(currentZoneId: TZoneId, adjacentZoneIds: TZoneId[]): void {
    // 收集需要保留的资源
    const neededKeys = new Set<string>();

    // 关键资源
    this._criticalResources.forEach(r => neededKeys.add(r.key));

    // 当前Zone资源
    this._zoneResources.get(currentZoneId)?.forEach(r => neededKeys.add(r.key));

    // 相邻Zone资源
    adjacentZoneIds.forEach(zoneId => {
      this._zoneResources.get(zoneId)?.forEach(r => neededKeys.add(r.key));
    });

    // 卸载不需要的资源
    this._loadedResources.forEach(key => {
      if (!neededKeys.has(key)) {
        this._unloadResource(key);
        this._loadedResources.delete(key);
      }
    });
  }

  /**
   * 检查资源是否已加载
   */
  isLoaded(key: string): boolean {
    return this._loadedResources.has(key);
  }

  /**
   * 获取纹理
   */
  getTexture(key: string): Phaser.Textures.Texture | undefined {
    return this._scene.textures.get(key);
  }

  // ==================== 私有方法 ====================

  private async _loadResources(resources: IResourceEntry[]): Promise<void> {
    return new Promise((resolve) => {
      if (resources.length === 0) {
        resolve();
        return;
      }

      resources.forEach(resource => {
        switch (resource.type) {
          case ResourceType.IMAGE:
            this._scene.load.image(resource.key, resource.path);
            break;

          case ResourceType.SPRITESHEET:
            this._scene.load.spritesheet(
              resource.key,
              resource.path,
              resource.frameConfig
            );
            break;

          case ResourceType.AUDIO:
            this._scene.load.audio(resource.key, resource.path);
            break;

          case ResourceType.YAML:
            this._scene.load.text(resource.key, resource.path);
            break;
        }
      });

      this._scene.load.once('complete', () => {
        resolve();
      });

      this._scene.load.start();
    });
  }

  private _unloadResource(key: string): void {
    // 卸载纹理
    if (this._scene.textures.exists(key)) {
      this._scene.textures.remove(key);
    }

    // 卸载音频
    if (this._scene.cache.audio.exists(key)) {
      this._scene.cache.audio.remove(key);
    }

    // 卸载YAML
    if (this._scene.cache.text.exists(key)) {
      this._scene.cache.text.remove(key);
    }
  }
}
```

### 5.1 完整实现

```typescript
// src/systems/scene/TransitionManager.ts

import Phaser from 'phaser';

/**
 * 过渡类型
 */
export enum TransitionType {
  NONE = 'none',
  FADE = 'fade',
  SLIDE_LEFT = 'slide_left',
  SLIDE_RIGHT = 'slide_right',
  SLIDE_UP = 'slide_up',
  SLIDE_DOWN = 'slide_down',
}

/**
 * 过渡管理器
 */
export class TransitionManager {
  private _scene: Phaser.Scene;
  private _overlay: Phaser.GameObjects.Rectangle | null = null;
  private _isTransitioning: boolean = false;

  constructor(scene: Phaser.Scene) {
    this._scene = scene;
  }

  /**
   * 是否正在过渡
   */
  isTransitioning(): boolean {
    return this._isTransitioning;
  }

  /**
   * 过渡出（淡出/滑出）
   */
  async out(type: TransitionType, duration: number): Promise<void> {
    if (type === TransitionType.NONE) return;

    this._isTransitioning = true;
    this._createOverlay();

    return new Promise(resolve => {
      switch (type) {
        case TransitionType.FADE:
          this._fadeOut(duration, resolve);
          break;

        case TransitionType.SLIDE_LEFT:
        case TransitionType.SLIDE_RIGHT:
        case TransitionType.SLIDE_UP:
        case TransitionType.SLIDE_DOWN:
          this._slideOut(type, duration, resolve);
          break;

        default:
          resolve();
      }
    });
  }

  /**
   * 过渡入（淡入/滑入）
   */
  async in(type: TransitionType, duration: number): Promise<void> {
    if (type === TransitionType.NONE) {
      this._destroyOverlay();
      return;
    }

    return new Promise(resolve => {
      switch (type) {
        case TransitionType.FADE:
          this._fadeIn(duration, resolve);
          break;

        case TransitionType.SLIDE_LEFT:
        case TransitionType.SLIDE_RIGHT:
        case TransitionType.SLIDE_UP:
        case TransitionType.SLIDE_DOWN:
          this._slideIn(type, duration, resolve);
          break;

        default:
          this._destroyOverlay();
          resolve();
      }
    });
  }

  // ==================== 私有方法 ====================

  private _createOverlay(): void {
    if (this._overlay) return;

    const { width, height } = this._scene.cameras.main;
    
    this._overlay = this._scene.add.rectangle(
      width / 2,
      height / 2,
      width,
      height,
      0x000000
    );
    
    this._overlay.setDepth(100000);
    this._overlay.setAlpha(0);
    this._overlay.setScrollFactor(0);
  }

  private _destroyOverlay(): void {
    if (this._overlay) {
      this._overlay.destroy();
      this._overlay = null;
    }
    this._isTransitioning = false;
  }

  private _fadeOut(duration: number, callback: () => void): void {
    if (!this._overlay) {
      callback();
      return;
    }

    this._scene.tweens.add({
      targets: this._overlay,
      alpha: 1,
      duration,
      ease: 'Power2',
      onComplete: () => callback(),
    });
  }

  private _fadeIn(duration: number, callback: () => void): void {
    if (!this._overlay) {
      callback();
      return;
    }

    this._overlay.setAlpha(1);

    this._scene.tweens.add({
      targets: this._overlay,
      alpha: 0,
      duration,
      ease: 'Power2',
      onComplete: () => {
        this._destroyOverlay();
        callback();
      },
    });
  }

  private _slideOut(type: TransitionType, duration: number, callback: () => void): void {
    if (!this._overlay) {
      callback();
      return;
    }

    const { width, height } = this._scene.cameras.main;
    let startX = width / 2;
    let startY = height / 2;
    let endX = width / 2;
    let endY = height / 2;

    // 根据方向设置起始和结束位置
    switch (type) {
      case TransitionType.SLIDE_LEFT:
        startX = -width / 2;
        break;
      case TransitionType.SLIDE_RIGHT:
        startX = width * 1.5;
        break;
      case TransitionType.SLIDE_UP:
        startY = -height / 2;
        break;
      case TransitionType.SLIDE_DOWN:
        startY = height * 1.5;
        break;
    }

    this._overlay.setPosition(startX, startY);
    this._overlay.setAlpha(1);

    this._scene.tweens.add({
      targets: this._overlay,
      x: endX,
      y: endY,
      duration,
      ease: 'Power2',
      onComplete: () => callback(),
    });
  }

  private _slideIn(type: TransitionType, duration: number, callback: () => void): void {
    if (!this._overlay) {
      callback();
      return;
    }

    const { width, height } = this._scene.cameras.main;
    let endX = width / 2;
    let endY = height / 2;

    // 根据方向设置结束位置
    switch (type) {
      case TransitionType.SLIDE_LEFT:
        endX = width * 1.5;
        break;
      case TransitionType.SLIDE_RIGHT:
        endX = -width / 2;
        break;
      case TransitionType.SLIDE_UP:
        endY = height * 1.5;
        break;
      case TransitionType.SLIDE_DOWN:
        endY = -height / 2;
        break;
    }

    this._scene.tweens.add({
      targets: this._overlay,
      x: endX,
      y: endY,
      duration,
      ease: 'Power2',
      onComplete: () => {
        this._destroyOverlay();
        callback();
      },
    });
  }
}
```
