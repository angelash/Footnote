## 1. 系统概述

### 1.1 UI层级定义

| 层级 | 深度值 | 说明 | 组件 |
|------|--------|------|------|
| **游戏层** | 0-999 | 游戏对象 | 背景、角色、物品 |
| **HUD层** | 1000-1999 | 常驻信息 | 交互提示、Zone标题 |
| **对话层** | 2000-2999 | 对话界面 | DialogueUI |
| **面板层** | 3000-3999 | 功能面板 | 卡片、地图、设置 |
| **菜单层** | 4000-4999 | 菜单界面 | 暂停、主菜单 |
| **模态层** | 5000-5999 | 模态框 | 确认框、警告 |
| **Toast层** | 6000-6999 | 提示信息 | 获得卡片、成就 |
| **过渡层** | 9000-9999 | 场景过渡 | 黑屏、加载 |

### 2.1 UIManager实现

```typescript
// src/systems/ui/UIManager.ts

import Phaser from 'phaser';
import { UILayer, UIPanelType, IUIManagerState } from '@/types';
import { DialogueUI } from './DialogueUI';
import { CardUI } from './CardUI';
import { MenuUI } from './MenuUI';
import { HUD } from './HUD';
import { ToastManager } from './ToastManager';
import { eventBus, GameEvent } from '@/systems/EventBus';

/**
 * UI管理器
 */
export class UIManager {
  private _scene: Phaser.Scene;
  private _state: IUIManagerState;
  private _dialogueUI: DialogueUI;
  private _cardUI: CardUI;
  private _menuUI: MenuUI;
  private _hud: HUD;
  private _toastManager: ToastManager;
  private _panels: Map<UIPanelType, BaseUIPanel> = new Map();
  private _inputLocked: boolean = false;

  constructor(scene: Phaser.Scene) {
    this._scene = scene;
    this._state = this._createInitialState();

    // 初始化UI组件
    this._dialogueUI = new DialogueUI(scene);
    this._cardUI = new CardUI(scene);
    this._menuUI = new MenuUI(scene);
    this._hud = new HUD(scene);
    this._toastManager = new ToastManager(scene);

    // 注册面板
    this._panels.set(UIPanelType.DIALOGUE, this._dialogueUI);
    this._panels.set(UIPanelType.CARD, this._cardUI);
    this._panels.set(UIPanelType.PAUSE, this._menuUI);

    this._setupEventListeners();
  }

  /**
   * 初始化
   */
  initialize(): void {
    this._hud.create();
    this._dialogueUI.create();
    this._cardUI.create();
    this._menuUI.create();
  }

  /**
   * 更新
   */
  update(delta: number): void {
    this._hud.update(delta);
  }

  // ==================== 对话UI ====================

  /**
   * 显示对话
   */
  showDialogue(speaker: string, text: string, portrait?: string): void {
    this._dialogueUI.show({ speaker, text, portrait });
    this._lockInput();
    this._state.openPanels.push(UIPanelType.DIALOGUE);
  }

  /**
   * 更新对话文本
   */
  updateDialogueText(text: string): void {
    this._dialogueUI.updateText(text);
  }

  /**
   * 显示对话选项
   */
  showDialogueOptions(options: Array<{ id: string; text: string }>): void {
    this._dialogueUI.showOptions(options);
  }

  /**
   * 隐藏对话
   */
  hideDialogue(): void {
    this._dialogueUI.hide();
    this._unlockInput();
    this._removeFromOpenPanels(UIPanelType.DIALOGUE);
  }

  /**
   * 设置对话回调
   */
  setDialogueCallbacks(callbacks: {
    onAdvance?: () => void;
    onChoice?: (choiceId: string) => void;
    onSkip?: () => void;
  }): void {
    this._dialogueUI.setCallbacks(callbacks);
  }

  // ==================== 卡片UI ====================

  /**
   * 显示卡片详情
   */
  showCard(cardData: ICard): void {
    this._cardUI.showCard(cardData);
    this._lockInput();
    this._state.openPanels.push(UIPanelType.CARD);
  }

  /**
   * 隐藏卡片
   */
  hideCard(): void {
    this._cardUI.hide();
    this._unlockInput();
    this._removeFromOpenPanels(UIPanelType.CARD);
  }

  /**
   * 显示卡片收藏
   */
  showCardCollection(): void {
    this._cardUI.showCollection();
    this._lockInput();
    this._state.openPanels.push(UIPanelType.CARD_COLLECTION);
  }

  // ==================== 菜单UI ====================

  /**
   * 显示暂停菜单
   */
  showPauseMenu(): void {
    this._menuUI.showPause();
    this._lockInput();
    this._state.openPanels.push(UIPanelType.PAUSE);
  }

  /**
   * 隐藏暂停菜单
   */
  hidePauseMenu(): void {
    this._menuUI.hide();
    this._unlockInput();
    this._removeFromOpenPanels(UIPanelType.PAUSE);
  }

  /**
   * 显示设置
   */
  showSettings(): void {
    this._menuUI.showSettings();
    this._state.openPanels.push(UIPanelType.SETTINGS);
  }

  // ==================== HUD ====================

  /**
   * 更新交互提示
   */
  updateInteractPrompt(text: string | null): void {
    this._hud.setInteractPrompt(text);
  }

  /**
   * 显示Zone标题
   */
  showZoneTitle(title: string, duration: number = 2000): void {
    this._hud.showZoneTitle(title, duration);
  }

  // ==================== Toast ====================

  /**
   * 显示Toast
   */
  showToast(message: string, type: 'info' | 'success' | 'warning' | 'error' = 'info'): void {
    this._toastManager.show(message, type);
  }

  /**
   * 显示卡片获得提示
   */
  showCardObtainedToast(cardTitle: string): void {
    this._toastManager.show(`获得卡片: ${cardTitle}`, 'success');
  }

  // ==================== 通用方法 ====================

  /**
   * 打开面板
   */
  openPanel(type: UIPanelType): void {
    const panel = this._panels.get(type);
    if (panel) {
      panel.show();
      this._state.openPanels.push(type);
      this._lockInput();
      
      eventBus.emit(GameEvent.UI_PANEL_OPEN, { panelType: type });
    }
  }

  /**
   * 关闭面板
   */
  closePanel(type: UIPanelType): void {
    const panel = this._panels.get(type);
    if (panel) {
      panel.hide();
      this._removeFromOpenPanels(type);
      
      if (this._state.openPanels.length === 0) {
        this._unlockInput();
      }
      
      eventBus.emit(GameEvent.UI_PANEL_CLOSE, { panelType: type });
    }
  }

  /**
   * 关闭所有面板
   */
  closeAllPanels(): void {
    for (const type of this._state.openPanels) {
      const panel = this._panels.get(type);
      panel?.hide();
    }
    this._state.openPanels = [];
    this._unlockInput();
  }

  /**
   * 是否有面板打开
   */
  hasOpenPanels(): boolean {
    return this._state.openPanels.length > 0;
  }

  /**
   * 输入是否被锁定
   */
  isInputLocked(): boolean {
    return this._inputLocked;
  }

  /**
   * 获取状态
   */
  getState(): IUIManagerState {
    return { ...this._state };
  }

  // ==================== 私有方法 ====================

  private _setupEventListeners(): void {
    // 监听Toast请求
    eventBus.on(GameEvent.UI_TOAST, (data: { message: string; type?: string }) => {
      this.showToast(data.message, data.type as any || 'info');
    });

    // 监听卡片获得
    eventBus.on(GameEvent.CARD_OBTAIN, (data: { cardId: string; card: ICard }) => {
      this.showCardObtainedToast(data.card.title);
    });

    // 监听Zone进入
    eventBus.on(GameEvent.ZONE_ENTER, (data: { zoneId: TZoneId }) => {
      const config = getZoneConfig(data.zoneId);
      if (config?.title) {
        this.showZoneTitle(config.title);
      }
    });
  }

  private _lockInput(): void {
    this._inputLocked = true;
    eventBus.emit(GameEvent.INPUT_LOCK);
  }

  private _unlockInput(): void {
    this._inputLocked = false;
    eventBus.emit(GameEvent.INPUT_UNLOCK);
  }

  private _removeFromOpenPanels(type: UIPanelType): void {
    const index = this._state.openPanels.indexOf(type);
    if (index > -1) {
      this._state.openPanels.splice(index, 1);
    }
  }

  private _createInitialState(): IUIManagerState {
    return {
      openPanels: [],
      activeToasts: [],
      inputLocked: false,
      isTransitioning: false,
    };
  }

  /**
   * 销毁
   */
  destroy(): void {
    this._dialogueUI.destroy();
    this._cardUI.destroy();
    this._menuUI.destroy();
    this._hud.destroy();
    this._toastManager.destroy();
  }
}
```

### 3.1 DialogueUI实现

```typescript
// src/systems/ui/DialogueUI.ts

import Phaser from 'phaser';
import { IDialogueUIConfig, UILayer } from '@/types';

/**
 * 默认对话UI配置
 */
const DEFAULT_CONFIG: IDialogueUIConfig = {
  background: 'ui_dialogue_bg',
  position: { x: 0, y: 900 },
  size: { width: 750, height: 380 },
  nameArea: {
    position: { x: 60, y: 920 },
    style: {
      fontSize: '28px',
      fontFamily: 'sans-serif',
      color: '#4A90D9',
      fontStyle: 'bold',
    },
  },
  textArea: {
    position: { x: 60, y: 980 },
    size: { width: 630, height: 200 },
    style: {
      fontSize: '26px',
      fontFamily: 'sans-serif',
      color: '#FFFFFF',
      wordWrap: { width: 630 },
    },
    lineSpacing: 8,
  },
  portraitArea: {
    position: { x: 150, y: 700 },
    size: { width: 200, height: 300 },
    origin: [0.5, 1],
  },
  optionsArea: {
    position: { x: 375, y: 1050 },
    spacing: 60,
    buttonConfig: {
      size: { width: 600, height: 50 },
    },
  },
  continuePrompt: {
    position: { x: 680, y: 1200 },
    texture: 'ui_continue_arrow',
  },
};

/**
 * 对话UI
 */
export class DialogueUI {
  private _scene: Phaser.Scene;
  private _config: IDialogueUIConfig;
  private _container: Phaser.GameObjects.Container | null = null;
  private _background: Phaser.GameObjects.NineSlice | null = null;
  private _nameText: Phaser.GameObjects.Text | null = null;
  private _dialogueText: Phaser.GameObjects.Text | null = null;
  private _portrait: Phaser.GameObjects.Image | null = null;
  private _continuePrompt: Phaser.GameObjects.Image | null = null;
  private _optionButtons: Phaser.GameObjects.Container[] = [];
  
  private _isVisible: boolean = false;
  private _isTyping: boolean = false;
  private _currentText: string = '';
  private _displayedText: string = '';
  private _typingTimer: Phaser.Time.TimerEvent | null = null;
  private _charIndex: number = 0;
  private _typingSpeed: number = 30; // 字/秒

  private _callbacks: {
    onAdvance?: () => void;
    onChoice?: (choiceId: string) => void;
    onSkip?: () => void;
  } = {};

  constructor(scene: Phaser.Scene, config: Partial<IDialogueUIConfig> = {}) {
    this._scene = scene;
    this._config = { ...DEFAULT_CONFIG, ...config };
  }

  /**
   * 创建UI元素
   */
  create(): void {
    this._container = this._scene.add.container(0, 0);
    this._container.setDepth(UILayer.DIALOGUE);
    this._container.setVisible(false);
    this._container.setScrollFactor(0);

    // 背景
    this._background = this._scene.add.nineslice(
      this._config.position.x + this._config.size.width / 2,
      this._config.position.y + this._config.size.height / 2,
      this._config.background,
      undefined,
      this._config.size.width,
      this._config.size.height,
      20, 20, 20, 20
    );
    this._background.setAlpha(0.95);
    this._container.add(this._background);

    // 名字
    this._nameText = this._scene.add.text(
      this._config.nameArea.position.x,
      this._config.nameArea.position.y,
      '',
      this._config.nameArea.style
    );
    this._container.add(this._nameText);

    // 对话文本
    this._dialogueText = this._scene.add.text(
      this._config.textArea.position.x,
      this._config.textArea.position.y,
      '',
      this._config.textArea.style
    );
    this._dialogueText.setLineSpacing(this._config.textArea.lineSpacing);
    this._container.add(this._dialogueText);

    // 立绘
    this._portrait = this._scene.add.image(
      this._config.portraitArea.position.x,
      this._config.portraitArea.position.y,
      'placeholder_portrait'
    );
    this._portrait.setOrigin(
      this._config.portraitArea.origin[0],
      this._config.portraitArea.origin[1]
    );
    this._portrait.setVisible(false);
    this._container.add(this._portrait);

    // 继续提示
    this._continuePrompt = this._scene.add.image(
      this._config.continuePrompt.position.x,
      this._config.continuePrompt.position.y,
      this._config.continuePrompt.texture
    );
    this._continuePrompt.setVisible(false);
    this._container.add(this._continuePrompt);

    // 闪烁动画
    this._scene.tweens.add({
      targets: this._continuePrompt,
      alpha: { from: 1, to: 0.3 },
      duration: 500,
      yoyo: true,
      repeat: -1,
    });

    // 点击推进
    this._scene.input.on('pointerdown', () => {
      if (!this._isVisible) return;
      
      if (this._isTyping) {
        this._completeTyping();
      } else if (this._optionButtons.length === 0) {
        this._callbacks.onAdvance?.();
      }
    });
  }

  /**
   * 显示对话
   */
  show(data: { speaker: string; text: string; portrait?: string }): void {
    if (!this._container) return;

    this._container.setVisible(true);
    this._isVisible = true;

    // 设置名字
    this._nameText?.setText(data.speaker);

    // 设置立绘
    if (data.portrait && this._scene.textures.exists(data.portrait)) {
      this._portrait?.setTexture(data.portrait);
      this._portrait?.setVisible(true);
    } else {
      this._portrait?.setVisible(false);
    }

    // 开始打字效果
    this._startTyping(data.text);

    // 显示进入动画
    this._playShowAnimation();
  }

  /**
   * 更新文本
   */
  updateText(text: string): void {
    this._startTyping(text);
  }

  /**
   * 显示选项
   */
  showOptions(options: Array<{ id: string; text: string }>): void {
    this._clearOptions();
    this._continuePrompt?.setVisible(false);

    const startY = this._config.optionsArea.position.y;
    const spacing = this._config.optionsArea.spacing;

    options.forEach((option, index) => {
      const y = startY + index * spacing;
      const button = this._createOptionButton(option, y);
      this._optionButtons.push(button);
      this._container?.add(button);
    });
  }

  /**
   * 隐藏
   */
  hide(): void {
    if (!this._container) return;

    this._playHideAnimation(() => {
      this._container?.setVisible(false);
      this._isVisible = false;
      this._clearOptions();
      this._stopTyping();
    });
  }

  /**
   * 设置回调
   */
  setCallbacks(callbacks: typeof this._callbacks): void {
    this._callbacks = callbacks;
  }

  /**
   * 销毁
   */
  destroy(): void {
    this._stopTyping();
    this._container?.destroy();
  }

  // ==================== 私有方法 ====================

  private _startTyping(text: string): void {
    this._stopTyping();

    this._currentText = text;
    this._displayedText = '';
    this._charIndex = 0;
    this._isTyping = true;
    this._continuePrompt?.setVisible(false);

    const interval = 1000 / this._typingSpeed;

    this._typingTimer = this._scene.time.addEvent({
      delay: interval,
      callback: this._typeNextChar,
      callbackScope: this,
      loop: true,
    });
  }

  private _typeNextChar(): void {
    if (this._charIndex >= this._currentText.length) {
      this._completeTyping();
      return;
    }

    this._displayedText += this._currentText[this._charIndex];
    this._dialogueText?.setText(this._displayedText);
    this._charIndex++;
  }

  private _completeTyping(): void {
    this._stopTyping();
    this._displayedText = this._currentText;
    this._dialogueText?.setText(this._displayedText);
    this._isTyping = false;
    this._continuePrompt?.setVisible(true);
  }

  private _stopTyping(): void {
    if (this._typingTimer) {
      this._typingTimer.destroy();
      this._typingTimer = null;
    }
  }

  private _createOptionButton(
    option: { id: string; text: string },
    y: number
  ): Phaser.GameObjects.Container {
    const container = this._scene.add.container(this._config.optionsArea.position.x, y);

    const bg = this._scene.add.rectangle(
      0, 0,
      this._config.optionsArea.buttonConfig.size!.width,
      this._config.optionsArea.buttonConfig.size!.height,
      0x333333
    );
    bg.setStrokeStyle(2, 0x666666);

    const text = this._scene.add.text(0, 0, option.text, {
      fontSize: '24px',
      fontFamily: 'sans-serif',
      color: '#FFFFFF',
    });
    text.setOrigin(0.5, 0.5);

    container.add([bg, text]);

    // 交互
    bg.setInteractive({ useHandCursor: true });

    bg.on('pointerover', () => {
      bg.setFillStyle(0x555555);
    });

    bg.on('pointerout', () => {
      bg.setFillStyle(0x333333);
    });

    bg.on('pointerdown', () => {
      this._callbacks.onChoice?.(option.id);
    });

    return container;
  }

  private _clearOptions(): void {
    this._optionButtons.forEach(btn => btn.destroy());
    this._optionButtons = [];
  }

  private _playShowAnimation(): void {
    if (!this._container) return;

    this._container.setAlpha(0);
    this._container.setY(50);

    this._scene.tweens.add({
      targets: this._container,
      alpha: 1,
      y: 0,
      duration: 200,
      ease: 'Power2',
    });
  }

  private _playHideAnimation(onComplete: () => void): void {
    if (!this._container) {
      onComplete();
      return;
    }

    this._scene.tweens.add({
      targets: this._container,
      alpha: 0,
      y: 50,
      duration: 150,
      ease: 'Power2',
      onComplete,
    });
  }
}
```

```typescript
// src/systems/ui/CardUI.ts

import Phaser from 'phaser';
import { ICard, CardCategory, UILayer } from '@/types';

/**
 * 卡片UI
 */
export class CardUI {
  private _scene: Phaser.Scene;
  private _container: Phaser.GameObjects.Container | null = null;
  private _overlay: Phaser.GameObjects.Rectangle | null = null;
  private _cardContainer: Phaser.GameObjects.Container | null = null;
  private _isVisible: boolean = false;
  private _onClose?: () => void;

  constructor(scene: Phaser.Scene) {
    this._scene = scene;
  }

  /**
   * 创建
   */
  create(): void {
    const { width, height } = this._scene.cameras.main;

    this._container = this._scene.add.container(0, 0);
    this._container.setDepth(UILayer.PANEL);
    this._container.setVisible(false);
    this._container.setScrollFactor(0);

    // 半透明遮罩
    this._overlay = this._scene.add.rectangle(
      width / 2,
      height / 2,
      width,
      height,
      0x000000,
      0.7
    );
    this._overlay.setInteractive();
    this._overlay.on('pointerdown', () => this.hide());
    this._container.add(this._overlay);

    // 卡片容器
    this._cardContainer = this._scene.add.container(width / 2, height / 2);
    this._container.add(this._cardContainer);
  }

  /**
   * 显示卡片详情
   */
  showCard(card: ICard): void {
    if (!this._container || !this._cardContainer) return;

    this._clearCard();

    // 卡片背景
    const cardBg = this._scene.add.rectangle(0, 0, 400, 560, 0x1a1a2e);
    cardBg.setStrokeStyle(3, this._getCategoryColor(card.category));
    this._cardContainer.add(cardBg);

    // 卡片图片区域
    const imageArea = this._scene.add.rectangle(0, -150, 360, 200, 0x2a2a4e);
    this._cardContainer.add(imageArea);

    if (card.image && this._scene.textures.exists(card.image)) {
      const cardImage = this._scene.add.image(0, -150, card.image);
      cardImage.setDisplaySize(360, 200);
      this._cardContainer.add(cardImage);
    }

    // 类别标签
    const categoryLabel = this._scene.add.text(-170, -260, this._getCategoryLabel(card.category), {
      fontSize: '16px',
      fontFamily: 'sans-serif',
      color: '#FFFFFF',
      backgroundColor: this._getCategoryColorHex(card.category),
      padding: { x: 8, y: 4 },
    });
    this._cardContainer.add(categoryLabel);

    // 标题
    const title = this._scene.add.text(0, -30, card.title, {
      fontSize: '28px',
      fontFamily: 'sans-serif',
      color: '#FFFFFF',
      fontStyle: 'bold',
    });
    title.setOrigin(0.5, 0.5);
    this._cardContainer.add(title);

    // 副标题
    if (card.subtitle) {
      const subtitle = this._scene.add.text(0, 10, card.subtitle, {
        fontSize: '18px',
```

```typescript
    });
    this._zoneTitle.setOrigin(0.5, 0.5);
    this._zoneTitle.setAlpha(0);
    this._container.add(this._zoneTitle);

    // 调试信息
    if (this._showDebug) {
      this._debugInfo = this._scene.add.text(10, 10, '', {
        fontSize: '14px',
        fontFamily: 'monospace',
        color: '#00FF00',
        backgroundColor: 'rgba(0, 0, 0, 0.5)',
        padding: { x: 5, y: 5 },
      });
      this._container.add(this._debugInfo);
    }
  }

  /**
   * 更新
   */
  update(delta: number): void {
    if (this._showDebug && this._debugInfo) {
      this._updateDebugInfo();
    }
  }

  /**
   * 设置交互提示
   */
  setInteractPrompt(text: string | null): void {
    if (!this._interactPrompt) return;

    if (text) {
      const textObj = this._interactPrompt.getAt(1) as Phaser.GameObjects.Text;
      textObj.setText(text);
      this._interactPrompt.setVisible(true);
    } else {
      this._interactPrompt.setVisible(false);
    }
  }

  /**
   * 显示Zone标题
   */
  showZoneTitle(title: string, duration: number = 2000): void {
    if (!this._zoneTitle) return;

    this._zoneTitle.setText(title);

    this._scene.tweens.add({
      targets: this._zoneTitle,
      alpha: { from: 0, to: 1 },
      duration: 300,
      ease: 'Power2',
      onComplete: () => {
        this._scene.time.delayedCall(duration, () => {
          this._scene.tweens.add({
            targets: this._zoneTitle,
            alpha: 0,
            duration: 300,
            ease: 'Power2',
          });
        });
      },
    });
  }

  /**
   * 切换调试信息
   */
  toggleDebug(): void {
    this._showDebug = !this._showDebug;
    this._debugInfo?.setVisible(this._showDebug);
  }

  /**
   * 销毁
   */
  destroy(): void {
    this._container?.destroy();
  }

  // ==================== 私有方法 ====================

  private _createInteractPrompt(): void {
    const { width, height } = this._scene.cameras.main;

    this._interactPrompt = this._scene.add.container(width / 2, height - 150);
    this._interactPrompt.setVisible(false);

    const bg = this._scene.add.rectangle(0, 0, 200, 40, 0x000000, 0.7);
    bg.setStrokeStyle(1, 0x666666);

    const text = this._scene.add.text(0, 0, '', {

      `Zone: ${worldState.getCurrentZone() || 'N/A'}`,
    ].join('\n'));
  }
}
```

### 6.1 ToastManager实现

```typescript
// src/systems/ui/ToastManager.ts

import Phaser from 'phaser';
import { UILayer } from '@/types';

/**
 * Toast类型配置
 */
const TOAST_STYLES = {
  info: { bg: 0x333333, border: 0x666666 },
  success: { bg: 0x1a5a2e, border: 0x2ecc71 },
  warning: { bg: 0x5a4a1a, border: 0xf1c40f },
  error: { bg: 0x5a1a1a, border: 0xe74c3c },
};

/**
 * Toast管理器
 */
export class ToastManager {
  private _scene: Phaser.Scene;
  private _container: Phaser.GameObjects.Container | null = null;
  private _toasts: Phaser.GameObjects.Container[] = [];
  private _maxToasts: number = 3;
  private _duration: number = 3000;

  constructor(scene: Phaser.Scene) {
    this._scene = scene;
    this._create();
  }

  /**
   * 显示Toast
   */
  show(message: string, type: keyof typeof TOAST_STYLES = 'info'): void {
    const toast = this._createToast(message, type);
    
    // 移除多余的Toast
    while (this._toasts.length >= this._maxToasts) {
      const oldToast = this._toasts.shift();
      oldToast?.destroy();
    }

    this._toasts.push(toast);
    this._container?.add(toast);

    // 重新排列位置
    this._arrangeToasts();

    // 进入动画
    toast.setAlpha(0);
    toast.setX(100);
    this._scene.tweens.add({
      targets: toast,
      alpha: 1,
      x: 0,
      duration: 200,
      ease: 'Power2',
    });

    // 自动消失
    this._scene.time.delayedCall(this._duration, () => {
      this._hideToast(toast);
    });
  }

  /**
   * 销毁
   */
  destroy(): void {
    this._container?.destroy();
  }

  // ==================== 私有方法 ====================

  private _create(): void {
    const { width } = this._scene.cameras.main;

    this._container = this._scene.add.container(width / 2, 80);
    this._container.setDepth(UILayer.TOAST);
    this._container.setScrollFactor(0);
  }

  private _createToast(message: string, type: keyof typeof TOAST_STYLES): Phaser.GameObjects.Container {
    const style = TOAST_STYLES[type];
    const container = this._scene.add.container(0, 0);

    const bg = this._scene.add.rectangle(0, 0, 0, 44, style.bg);
    bg.setStrokeStyle(2, style.border);

    const text = this._scene.add.text(0, 0, message, {
      fontSize: '20px',
      fontFamily: 'sans-serif',
      color: '#FFFFFF',
    });
    text.setOrigin(0.5, 0.5);

    // 调整背景宽度
    bg.setSize(text.width + 40, 44);

    container.add([bg, text]);

    return container;
  }

  private _hideToast(toast: Phaser.GameObjects.Container): void {
    const index = this._toasts.indexOf(toast);
    if (index > -1) {
      this._toasts.splice(index, 1);
    }

    this._scene.tweens.add({
      targets: toast,
      alpha: 0,
      x: -100,
      duration: 200,
      ease: 'Power2',
      onComplete: () => {
        toast.destroy();
        this._arrangeToasts();
      },
    });
  }

  private _arrangeToasts(): void {
    this._toasts.forEach((toast, index) => {
      this._scene.tweens.add({
        targets: toast,
        y: index * 60,
        duration: 150,
        ease: 'Power2',
      });
    });
  }
}
```
