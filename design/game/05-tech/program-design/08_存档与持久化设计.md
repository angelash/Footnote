## 1. 系统概述

### 1.1 设计目标

| 目标 | 说明 |
|------|------|
| **可靠性** | 存档数据不丢失，支持故障恢复 |
| **透明性** | 自动存档，玩家无感知 |
| **兼容性** | 跨版本存档迁移支持 |
| **安全性** | 防止存档篡改，可校验完整性 |

### 1.2 技术选型

| 方案 | 技术 | 说明 |
|------|------|------|
| **主存储** | IndexedDB | 大容量、异步、结构化 |
| **备份存储** | LocalStorage | 小容量快照备份 |
| **云同步** | REST API | 可选的云端同步 |

## 3. SaveManager核心实现

### 3.1 SaveManager

```typescript
// src/systems/save/SaveManager.ts

import {
  ISaveData,
  ISaveMeta,
  ISaveSnapshot,
  TSaveSlot,
  SAVE_VERSION,
} from '@/types';
import { IndexedDBProvider } from './providers/IndexedDBProvider';
import { LocalStorageProvider } from './providers/LocalStorageProvider';
import { SaveEncoder } from './SaveEncoder';
import { SaveDecoder } from './SaveDecoder';
import { SaveValidator } from './SaveValidator';
import { eventBus, GameEvent } from '@/systems/EventBus';
import { worldState } from '@/systems/world/WorldState';

/**
 * 数据库配置
 */
const DB_CONFIG = {
  name: 'footnote_saves',
  version: 1,
  storeName: 'saves',
};

/**
 * 存档管理器
 */
export class SaveManager {
  private static _instance: SaveManager | null = null;
  private _dbProvider: IndexedDBProvider;
  private _backupProvider: LocalStorageProvider;
  private _encoder: SaveEncoder;
  private _decoder: SaveDecoder;
  private _validator: SaveValidator;
  private _initialized: boolean = false;
  private _playStartTime: number = 0;
  private _currentSlot: TSaveSlot = 'slot1';

  private constructor() {
    this._dbProvider = new IndexedDBProvider(DB_CONFIG);
    this._backupProvider = new LocalStorageProvider('footnote_backup');
    this._encoder = new SaveEncoder();
    this._decoder = new SaveDecoder();
    this._validator = new SaveValidator();
  }

  /**
   * 获取单例
   */
  static getInstance(): SaveManager {
    if (!SaveManager._instance) {
      SaveManager._instance = new SaveManager();
    }
    return SaveManager._instance;
  }

  /**
   * 初始化
   */
  async initialize(): Promise<void> {
    if (this._initialized) return;

    try {
      await this._dbProvider.open();
      this._initialized = true;
      this._playStartTime = Date.now();
      console.log('[SaveManager] Initialized');
    } catch (error) {
      console.error('[SaveManager] Failed to initialize:', error);
      throw error;
    }
  }

  // ==================== 存档操作 ====================

  /**
   * 保存游戏
   */
  async save(slot: TSaveSlot, name?: string): Promise<boolean> {
    try {
      const saveData = this._collectSaveData(slot, name);
      
      // 计算校验和
      saveData.checksum = this._encoder.calculateChecksum(saveData);

      // 编码存档
      const encoded = this._encoder.encode(saveData);

      // 写入主存储
      await this._dbProvider.write(slot, encoded);

      // 写入备份
      if (slot !== 'autosave') {
        this._backupProvider.write(`backup_${slot}`, encoded);
      }

      eventBus.emit(GameEvent.SAVE_COMPLETE, { slot });
      console.log(`[SaveManager] Saved to ${slot}`);
      return true;
    } catch (error) {
      console.error('[SaveManager] Save failed:', error);
      eventBus.emit(GameEvent.SAVE_ERROR, { slot, error });
      return false;
    }
  }

  /**
   * 加载存档
   */
  async load(slot: TSaveSlot): Promise<ISaveData | null> {
    try {
      // 读取主存储
      let encoded = await this._dbProvider.read(slot);

      // 主存储失败，尝试备份
      if (!encoded && slot !== 'autosave') {
        console.warn(`[SaveManager] Primary read failed, trying backup for ${slot}`);
        encoded = this._backupProvider.read(`backup_${slot}`);
      }

      if (!encoded) {
        console.log(`[SaveManager] No save found in ${slot}`);
        return null;
      }

      // 解码存档
      const saveData = this._decoder.decode(encoded);

      // 校验存档
      const validation = this._validator.validate(saveData);
      if (!validation.valid) {
        console.error('[SaveManager] Save validation failed:', validation.errors);
        return null;
      }

      // 迁移旧版本
      const migrated = this._migrateIfNeeded(saveData);

      // 应用存档
      this._applySaveData(migrated);

      this._currentSlot = slot;
      eventBus.emit(GameEvent.LOAD_COMPLETE, { slot });
      console.log(`[SaveManager] Loaded from ${slot}`);
      return migrated;
    } catch (error) {
      console.error('[SaveManager] Load failed:', error);
      eventBus.emit(GameEvent.LOAD_ERROR, { slot, error });
      return null;
    }
  }

  /**
   * 删除存档
   */
  async delete(slot: TSaveSlot): Promise<boolean> {
    try {
      await this._dbProvider.delete(slot);
      this._backupProvider.delete(`backup_${slot}`);
      
      eventBus.emit(GameEvent.SAVE_DELETE, { slot });
      console.log(`[SaveManager] Deleted ${slot}`);
      return true;
    } catch (error) {
      console.error('[SaveManager] Delete failed:', error);
      return false;
    }
  }

  /**
   * 获取所有存档快照
   */
  async getAllSnapshots(): Promise<ISaveSnapshot[]> {
    const slots: TSaveSlot[] = ['slot1', 'slot2', 'slot3', 'autosave', 'quicksave'];
    const snapshots: ISaveSnapshot[] = [];

    for (const slot of slots) {
      const encoded = await this._dbProvider.read(slot);
      
      if (encoded) {
        try {
          const saveData = this._decoder.decode(encoded);
          snapshots.push({
            slot,
            meta: saveData.meta,
            isEmpty: false,
          });
        } catch {
          snapshots.push({
            slot,
            meta: this._createEmptyMeta(slot),
            isEmpty: true,
          });
        }
      } else {
        snapshots.push({
          slot,
          meta: this._createEmptyMeta(slot),
          isEmpty: true,
        });
      }
    }

    return snapshots;
  }

  /**
   * 检查存档是否存在
   */
  async exists(slot: TSaveSlot): Promise<boolean> {
    const encoded = await this._dbProvider.read(slot);
    return encoded !== null;
  }

  // ==================== 快速存档 ====================

  /**
   * 快速保存
   */
  async quickSave(): Promise<boolean> {
    return this.save('quicksave', '快速存档');
  }

  /**
   * 快速加载
   */
  async quickLoad(): Promise<ISaveData | null> {
    return this.load('quicksave');
  }

  // ==================== 自动存档 ====================

  /**
   * 自动保存
   */
  async autoSave(): Promise<boolean> {
    return this.save('autosave', '自动存档');
  }

  // ==================== 导入导出 ====================

  /**
   * 导出存档为JSON字符串
   */
  async export(slot: TSaveSlot): Promise<string | null> {
    const encoded = await this._dbProvider.read(slot);
    if (!encoded) return null;

    const saveData = this._decoder.decode(encoded);
    return JSON.stringify(saveData, null, 2);
  }

  /**
   * 导入存档
   */
  async import(slot: TSaveSlot, jsonString: string): Promise<boolean> {
    try {
      const saveData = JSON.parse(jsonString) as ISaveData;
      
      // 校验
      const validation = this._validator.validate(saveData);
      if (!validation.valid) {
        console.error('[SaveManager] Import validation failed:', validation.errors);
        return false;
      }

      // 更新元数据
      saveData.meta.slot = slot;
      saveData.meta.updatedAt = Date.now();

      // 重新计算校验和
      saveData.checksum = this._encoder.calculateChecksum(saveData);

      // 保存
      const encoded = this._encoder.encode(saveData);
      await this._dbProvider.write(slot, encoded);

      return true;
    } catch (error) {
      console.error('[SaveManager] Import failed:', error);
      return false;
    }
  }

  // ==================== 工具方法 ====================

  /**
   * 获取当前游戏时长
   */
  getPlayTime(): number {
    return Date.now() - this._playStartTime;
  }

  /**
   * 获取当前槽位
   */
  getCurrentSlot(): TSaveSlot {
    return this._currentSlot;
  }

  // ==================== 私有方法 ====================

  /**
   * 收集存档数据
   */
  private _collectSaveData(slot: TSaveSlot, name?: string): ISaveData {
    const state = worldState.getState();
    const currentZone = worldState.getCurrentZone();
    const counters = worldState.getCounters();

    const meta: ISaveMeta = {
      version: SAVE_VERSION,
      slot,
      name: name || this._getDefaultName(slot),
      createdAt: Date.now(),
      updatedAt: Date.now(),
      playTime: this.getPlayTime(),
      chapter: this._getCurrentChapter(),
      zoneId: currentZone,
    };

    return {
      meta,
      worldState: {
        counters: {
          R: counters.R,
          P: counters.P,
          baseW: state.baseW,
        },
        abilities: [...state.abilities],
        flags: { ...state.flags },
        zoneStates: this._serializeZoneStates(state.zoneStates),
        scars: state.scars.map(s => ({ ...s })),
        contaminations: state.contaminations.map(c => ({ ...c })),
      },
      progress: this._collectProgress(),
      inventory: this._collectInventory(),
      settings: this._collectSettings(),
      checksum: '',
    };
  }

  /**
   * 应用存档数据
   */
  private _applySaveData(saveData: ISaveData): void {
    // 恢复世界状态
    worldState.restore({
      counters: saveData.worldState.counters,
      abilities: saveData.worldState.abilities,
      flags: saveData.worldState.flags,
      zoneStates: this._deserializeZoneStates(saveData.worldState.zoneStates),
      scars: saveData.worldState.scars,
      contaminations: saveData.worldState.contaminations,
    });

    // 恢复进度
    // progressManager.restore(saveData.progress);

    // 恢复收集品
    // inventoryManager.restore(saveData.inventory);

    // 恢复设置
    // settingsManager.restore(saveData.settings);

    // 恢复游戏时长
    this._playStartTime = Date.now() - saveData.meta.playTime;
  }

  /**
   * 版本迁移
   */
  private _migrateIfNeeded(saveData: ISaveData): ISaveData {
    const currentVersion = SAVE_VERSION;
    const saveVersion = saveData.meta.version;

    if (saveVersion === currentVersion) {
      return saveData;
    }

    console.log(`[SaveManager] Migrating save from ${saveVersion} to ${currentVersion}`);

    // 添加版本迁移逻辑
    // if (compareVersions(saveVersion, '1.1.0') < 0) {
    //   saveData = this._migrateToV110(saveData);
    // }

    saveData.meta.version = currentVersion;
    return saveData;
  }

  private _getDefaultName(slot: TSaveSlot): string {
    const names: Record<TSaveSlot, string> = {
      slot1: '存档 1',
      slot2: '存档 2',
      slot3: '存档 3',
      autosave: '自动存档',
      quicksave: '快速存档',
    };
    return names[slot];
  }

  private _getCurrentChapter(): string {
    const zoneId = worldState.getCurrentZone();
    if (!zoneId) return 'C0';
    return zoneId.split('-')[0];
  }

  private _serializeZoneStates(states: Map<TZoneId, IZoneState>): Record<TZoneId, IZoneStateSave> {
    const result: Record<string, IZoneStateSave> = {};
    states.forEach((state, id) => {
      result[id] = {
        unlocked: state.unlocked,
        completed: state.completed,
        visitCount: state.visitCount,
        firstVisitTime: state.firstVisitTime,
        completedTime: state.completedTime,
        collectedItems: [...state.collectedItems],
        triggeredEvents: [...state.triggeredEvents],
      };
    });
    return result as Record<TZoneId, IZoneStateSave>;
  }

  private _deserializeZoneStates(saved: Record<TZoneId, IZoneStateSave>): Map<TZoneId, IZoneState> {
    const result = new Map<TZoneId, IZoneState>();
    Object.entries(saved).forEach(([id, state]) => {
      result.set(id as TZoneId, {
        unlocked: state.unlocked,
        completed: state.completed,
        visitCount: state.visitCount,
        firstVisitTime: state.firstVisitTime,
        completedTime: state.completedTime,
        collectedItems: new Set(state.collectedItems),
        triggeredEvents: new Set(state.triggeredEvents),
      });
    });
    return result;
  }

  private _collectProgress(): IProgressSave {
    return {
      currentZoneId: worldState.getCurrentZone(),
      currentChapter: this._getCurrentChapter(),
      mainProgress: 0,
      completedZones: [],
      triggeredForeshadows: [],
      endingConditions: {},
      achievements: [],
    };
  }

  private _collectInventory(): IInventorySave {
    return {
      cards: [],
      viewedCards: [],
    };
  }

  private _collectSettings(): ISettingsSave {
    return {
      volume: {
        master: 1,
        bgm: 0.8,
        sfx: 1,
        voice: 1,
      },
      textSpeed: 1,
      autoPlayDelay: 2000,
      language: 'zh-CN',
      screenShake: true,
      accessibility: {
        highContrast: false,
        reduceMotion: false,
        fontSize: 1,
      },
    };
  }

  private _createEmptyMeta(slot: TSaveSlot): ISaveMeta {
    return {
      version: SAVE_VERSION,
      slot,
      name: this._getDefaultName(slot),
      createdAt: 0,
      updatedAt: 0,
      playTime: 0,
      chapter: '',
      zoneId: '' as TZoneId,
    };
  }
}

export const saveManager = SaveManager.getInstance();
```

```typescript
// src/systems/save/providers/IndexedDBProvider.ts

export interface IDBConfig {
  name: string;
  version: number;
  storeName: string;
}

/**
 * IndexedDB存储提供者
 */
export class IndexedDBProvider {
  private _config: IDBConfig;
  private _db: IDBDatabase | null = null;

  constructor(config: IDBConfig) {
    this._config = config;
  }

  /**
   * 打开数据库
   */
  async open(): Promise<void> {
    return new Promise((resolve, reject) => {
      const request = indexedDB.open(this._config.name, this._config.version);

      request.onerror = () => {
        reject(new Error(`Failed to open database: ${request.error}`));
      };

      request.onsuccess = () => {
        this._db = request.result;
        resolve();
      };

      request.onupgradeneeded = (event) => {
        const db = (event.target as IDBOpenDBRequest).result;
        
        if (!db.objectStoreNames.contains(this._config.storeName)) {
          db.createObjectStore(this._config.storeName, { keyPath: 'slot' });
        }
      };
    });
  }

  /**
   * 写入数据
   */
  async write(slot: string, data: string): Promise<void> {
    if (!this._db) throw new Error('Database not opened');

    return new Promise((resolve, reject) => {
      const transaction = this._db!.transaction([this._config.storeName], 'readwrite');
      const store = transaction.objectStore(this._config.storeName);
      
      const request = store.put({ slot, data, updatedAt: Date.now() });

      request.onerror = () => reject(new Error(`Write failed: ${request.error}`));
      request.onsuccess = () => resolve();
    });
  }

  /**
   * 读取数据
   */
  async read(slot: string): Promise<string | null> {
    if (!this._db) throw new Error('Database not opened');

    return new Promise((resolve, reject) => {
      const transaction = this._db!.transaction([this._config.storeName], 'readonly');
      const store = transaction.objectStore(this._config.storeName);
      
      const request = store.get(slot);

      request.onerror = () => reject(new Error(`Read failed: ${request.error}`));
      request.onsuccess = () => {
        const result = request.result;
        resolve(result ? result.data : null);
      };
    });
  }

  /**
   * 删除数据
   */
  async delete(slot: string): Promise<void> {
    if (!this._db) throw new Error('Database not opened');

    return new Promise((resolve, reject) => {
      const transaction = this._db!.transaction([this._config.storeName], 'readwrite');
      const store = transaction.objectStore(this._config.storeName);
      
      const request = store.delete(slot);

      request.onerror = () => reject(new Error(`Delete failed: ${request.error}`));
      request.onsuccess = () => resolve();
    });
  }

  /**
   * 关闭数据库
   */
  close(): void {
    this._db?.close();
    this._db = null;
  }
}
```

```typescript
// src/systems/save/providers/LocalStorageProvider.ts

/**
 * LocalStorage存储提供者
 */
export class LocalStorageProvider {
  private _prefix: string;

  constructor(prefix: string) {
    this._prefix = prefix;
  }

  /**
   * 写入数据
   */
  write(key: string, data: string): void {
    try {
      localStorage.setItem(`${this._prefix}_${key}`, data);
    } catch (error) {
      console.error('[LocalStorageProvider] Write failed:', error);
      // LocalStorage已满，尝试清理旧数据
      this._cleanup();
      localStorage.setItem(`${this._prefix}_${key}`, data);
    }
  }

  /**
   * 读取数据
   */
  read(key: string): string | null {
    return localStorage.getItem(`${this._prefix}_${key}`);
  }

  /**
   * 删除数据
   */
  delete(key: string): void {
    localStorage.removeItem(`${this._prefix}_${key}`);
  }

  /**
   * 清理旧数据
   */
  private _cleanup(): void {
    const keysToRemove: string[] = [];
    
    for (let i = 0; i < localStorage.length; i++) {
      const key = localStorage.key(i);
      if (key?.startsWith(this._prefix)) {
        keysToRemove.push(key);
      }
    }

    // 保留最新的3个
    if (keysToRemove.length > 3) {
      keysToRemove.slice(0, -3).forEach(key => {
        localStorage.removeItem(key);
      });
    }
  }
}
```

## 5. 编解码与校验

### 5.1 SaveEncoder

```typescript
// src/systems/save/SaveEncoder.ts

import { ISaveData } from '@/types';

/**
 * 存档编码器
 */
export class SaveEncoder {
  /**
   * 编码存档
   */
  encode(saveData: ISaveData): string {
    const json = JSON.stringify(saveData);
    // 简单的Base64编码
    return btoa(encodeURIComponent(json));
  }

  /**
   * 计算校验和
   */
  calculateChecksum(saveData: ISaveData): string {
    const data = { ...saveData, checksum: '' };
    const json = JSON.stringify(data);
    return this._simpleHash(json);
  }

  /**
   * 简单哈希算法
   */
  private _simpleHash(str: string): string {
    let hash = 0;
    for (let i = 0; i < str.length; i++) {
      const char = str.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash;
    }
    return hash.toString(16);
  }
}
```

### 5.3 SaveValidator

```typescript
// src/systems/save/SaveValidator.ts

import { ISaveData, SAVE_VERSION } from '@/types';
import { SaveEncoder } from './SaveEncoder';

export interface IValidationResult {
  valid: boolean;
  errors: string[];
}

/**
 * 存档校验器
 */
export class SaveValidator {
  private _encoder: SaveEncoder;

  constructor() {
    this._encoder = new SaveEncoder();
  }

  /**
   * 校验存档
   */
  validate(saveData: ISaveData): IValidationResult {
    const errors: string[] = [];

    // 校验基本结构
    if (!saveData.meta) errors.push('Missing meta');
    if (!saveData.worldState) errors.push('Missing worldState');
    if (!saveData.progress) errors.push('Missing progress');
    if (!saveData.inventory) errors.push('Missing inventory');
    if (!saveData.settings) errors.push('Missing settings');

    // 校验版本
    if (saveData.meta?.version) {
      const version = this._parseVersion(saveData.meta.version);
      const current = this._parseVersion(SAVE_VERSION);
      
      if (version.major > current.major) {
        errors.push(`Save version ${saveData.meta.version} is newer than current ${SAVE_VERSION}`);
      }
    }

    // 校验校验和
    if (saveData.checksum) {
      const expected = this._encoder.calculateChecksum(saveData);
      if (saveData.checksum !== expected) {
        errors.push('Checksum mismatch - save may be corrupted');
      }
    }

    // 校验计数器范围
    if (saveData.worldState?.counters) {
      const { R, P, baseW } = saveData.worldState.counters;
      if (R < 0 || R > 100) errors.push('Invalid R value');
      if (P < 0 || P > 100) errors.push('Invalid P value');
      if (baseW < 0 || baseW > 100) errors.push('Invalid W value');
    }

    return {
      valid: errors.length === 0,
      errors,
    };
  }

  private _parseVersion(version: string): { major: number; minor: number; patch: number } {
    const parts = version.split('.').map(Number);
    return {
      major: parts[0] || 0,
      minor: parts[1] || 0,
      patch: parts[2] || 0,
    };
  }
}
```

## 6. AutoSaveManager

### 6.1 自动存档管理器

```typescript
// src/systems/save/AutoSaveManager.ts

import { saveManager } from './SaveManager';
import { eventBus, GameEvent } from '@/systems/EventBus';

/**
 * 自动存档配置
 */
interface IAutoSaveConfig {
  /** 定时存档间隔(毫秒) */
  interval: number;
  /** 是否在关键点存档 */
  saveOnKeyPoints: boolean;
  /** 是否在退出时存档 */
  saveOnExit: boolean;
}

const DEFAULT_CONFIG: IAutoSaveConfig = {
  interval: 60000, // 1分钟
  saveOnKeyPoints: true,
  saveOnExit: true,
};

/**
 * 自动存档管理器
 */
export class AutoSaveManager {
  private _config: IAutoSaveConfig;
  private _intervalId: number | null = null;
  private _enabled: boolean = true;

  constructor(config: Partial<IAutoSaveConfig> = {}) {
    this._config = { ...DEFAULT_CONFIG, ...config };
  }

  /**
   * 启动自动存档
   */
  start(): void {
    this._setupInterval();
    this._setupKeyPointListeners();
    this._setupExitListener();
  }

  /**
   * 停止自动存档
   */
  stop(): void {
    if (this._intervalId !== null) {
      clearInterval(this._intervalId);
      this._intervalId = null;
    }
  }

  /**
   * 启用/禁用
   */
  setEnabled(enabled: boolean): void {
    this._enabled = enabled;
  }

  /**
   * 触发自动存档
   */
  async trigger(reason: string = 'auto'): Promise<void> {
    if (!this._enabled) return;

    console.log(`[AutoSave] Triggered: ${reason}`);
    await saveManager.autoSave();
  }

  // ==================== 私有方法 ====================

  private _setupInterval(): void {
    this._intervalId = window.setInterval(() => {
      this.trigger('interval');
    }, this._config.interval);
  }

  private _setupKeyPointListeners(): void {
    if (!this._config.saveOnKeyPoints) return;

    // Zone切换时自动存档
    eventBus.on(GameEvent.ZONE_COMPLETE, () => {
      this.trigger('zone_complete');
    });

    // 获得卡片时自动存档
    eventBus.on(GameEvent.CARD_OBTAIN, () => {
      this.trigger('card_obtain');
    });

    // 章节完成时自动存档
    eventBus.on(GameEvent.CHAPTER_COMPLETE, () => {
      this.trigger('chapter_complete');
    });

    // 能力解锁时自动存档
    eventBus.on(GameEvent.ABILITY_UNLOCK, () => {
      this.trigger('ability_unlock');
    });
  }

  private _setupExitListener(): void {
    if (!this._config.saveOnExit) return;

    // 页面关闭/刷新前存档
    window.addEventListener('beforeunload', () => {
      // 使用同步localStorage作为紧急备份
      this._emergencyBackup();
    });

    // 页面可见性变化时存档（移动端后台）
    document.addEventListener('visibilitychange', () => {
      if (document.visibilityState === 'hidden') {
        this.trigger('visibility_hidden');
      }
    });
  }

  private _emergencyBackup(): void {
    // 紧急备份到LocalStorage
    try {
      const state = worldState.getState();
      const backup = JSON.stringify({
        timestamp: Date.now(),
        currentZone: worldState.getCurrentZone(),
        counters: worldState.getCounters(),
      });
      localStorage.setItem('footnote_emergency_backup', backup);
    } catch (error) {
      console.error('[AutoSave] Emergency backup failed:', error);
    }
  }
}
```
