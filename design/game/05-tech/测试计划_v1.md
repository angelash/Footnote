### 2.2 测试用例示例

```typescript
// tests/unit/systems/WorldState.test.ts
import { describe, it, expect, beforeEach } from 'vitest';
import { WorldState } from '@/systems/world/WorldState';

describe('WorldState', () => {
  let worldState: WorldState;
  
  beforeEach(() => {
    worldState = new WorldState();
  });
  
  describe('R值管理', () => {
    it('初始R值应为0', () => {
      expect(worldState.getR()).toBe(0);
    });
    
    it('增加R值应正确计算', () => {
      worldState.addR(2);
      expect(worldState.getR()).toBe(2);
    });
    
    it('R值不应超过上限15', () => {
      worldState.addR(20);
      expect(worldState.getR()).toBe(15);
    });
    
    it('R值不应为负', () => {
      worldState.addR(-5);
      expect(worldState.getR()).toBe(0);
    });
  });
  
  describe('W值计算', () => {
    it('初始W值应为100', () => {
      expect(worldState.getW()).toBe(100);
    });
    
    it('W值应根据R和P正确计算', () => {
      worldState.addR(6);  // -18
      worldState.addP(8);  // -16
      // W = 100 - 18 - 16 = 66
      expect(worldState.getW()).toBe(66);
    });
    
    it('W值不应低于0', () => {
      worldState.addR(15);
      worldState.addP(20);
      expect(worldState.getW()).toBe(0);
    });
  });
  
  describe('阈值触发', () => {
    it('R≥3时应触发subtle_pause', () => {
      const callback = vi.fn();
      worldState.onThreshold('R_subtle_pause', callback);
      worldState.addR(3);
      expect(callback).toHaveBeenCalled();
    });
    
    it('R≥10时应触发model_rewrite', () => {
      const callback = vi.fn();
      worldState.onThreshold('R_model_rewrite', callback);
      worldState.addR(10);
      expect(callback).toHaveBeenCalled();
    });
  });
});
```

// tests/e2e/prologue.spec.ts
import { test, expect } from '@playwright/test';

test.describe('序章流程', () => {
  test.beforeEach(async ({ page }) => {
    await page.goto('http://localhost:5173');
    await page.waitForSelector('[data-testid="start-game"]');
  });
  
  test('应该能开始新游戏并进入C0-Z1', async ({ page }) => {
    await page.click('[data-testid="new-game"]');
    await page.waitForSelector('[data-testid="zone-c0-z1"]');
    expect(await page.textContent('[data-testid="zone-title"]')).toBe('宿舍走廊');
  });
  
  test('应该能与工具包交互', async ({ page }) => {
    await page.click('[data-testid="new-game"]');
    await page.waitForSelector('[data-testid="toolbox"]');
    await page.click('[data-testid="toolbox"]');
    await expect(page.locator('[data-testid="item-obtained"]')).toBeVisible();
  });
  
  test('应该能完成C0并进入C1', async ({ page }) => {
    // 完成C0-Z1到C0-Z4的完整流程
    await page.click('[data-testid="new-game"]');
    
    // C0-Z1
    await page.waitForSelector('[data-testid="zone-c0-z1"]');
    await page.click('[data-testid="toolbox"]');
    await page.click('[data-testid="exit-south"]');
    
    // C0-Z2
    await page.waitForSelector('[data-testid="zone-c0-z2"]');
    // ... 完成交互
    
    // 验证进入C1
    await page.waitForSelector('[data-testid="chapter-c1"]');
    expect(await page.textContent('[data-testid="chapter-title"]')).toContain('第1章');

});

test.describe('存档系统', () => {
